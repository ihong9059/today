### 핵심 실습 코드: 공장 신호등

```python
# factory_signal.py - Shield Board 실제 핀 배치 기준
import RPi.GPIO as GPIO
import time
import board, busio, adafruit_ahtx0

# Shield Board 핀 설정 (회로도 기준)
LED_RED = 17     # Pin 11 - 위험
LED_YELLOW = 27  # Pin 13 - 주의
LED_BLUE = 22    # Pin 15 - 정상
BUZZER = 5       # Pin 29 - 경고음
SWITCH = 4       # Pin 7  - 스위치

GPIO.setmode(GPIO.BCM)
for pin in [LED_RED, LED_YELLOW, LED_BLUE, BUZZER]:
    GPIO.setup(pin, GPIO.OUT)
GPIO.setup(SWITCH, GPIO.IN, pull_up_down=GPIO.PUD_UP)

# AHT20 센서 (I2C: SDA=Pin3, SCL=Pin5)
i2c = busio.I2C(board.SCL, board.SDA)
sensor = adafruit_ahtx0.AHTx0(i2c)

try:
    while True:
        temp = sensor.temperature
        humi = sensor.relative_humidity
        print(f"온도: {temp:.1f}C, 습도: {humi:.1f}%")

        # 신호등 로직
        GPIO.output(LED_BLUE, temp < 28)       # 정상
        GPIO.output(LED_YELLOW, 28 <= temp < 35)  # 주의
        GPIO.output(LED_RED, temp >= 35)        # 위험
        GPIO.output(BUZZER, temp >= 35)         # 경고음

        time.sleep(2)
except KeyboardInterrupt:
    GPIO.cleanup()
```

### 핵심 실습 코드: AI 이상탐지 + 보고서

```python
# ai_anomaly_report.py - 이상탐지 + Claude API 보고서
import numpy as np
import board, busio, adafruit_ahtx0
import RPi.GPIO as GPIO
import time, json, requests

# 센서 및 GPIO 초기화
i2c = busio.I2C(board.SCL, board.SDA)
sensor = adafruit_ahtx0.AHTx0(i2c)
LED_RED = 17
BUZZER = 5
GPIO.setmode(GPIO.BCM)
GPIO.setup(LED_RED, GPIO.OUT)
GPIO.setup(BUZZER, GPIO.OUT)

# 1단계: 정상 데이터 수집 (30초)
print("정상 데이터 수집 중...")
normal = [sensor.temperature for _ in range(30)]
mean, std = np.mean(normal), np.std(normal)
print(f"기준값: 평균 {mean:.1f}C, 표준편차 {std:.2f}")

# 2단계: 실시간 감시
history = []
while True:
    temp = sensor.temperature
    z_score = abs(temp - mean) / std if std > 0 else 0
    status = "이상!" if z_score > 2 else "정상"

    GPIO.output(LED_RED, z_score > 2)
    GPIO.output(BUZZER, z_score > 2)

    history.append({"temp": temp, "z": z_score, "status": status})
    print(f"온도: {temp:.1f}C, Z-score: {z_score:.2f} → {status}")

    # 이상 감지 시 AI 보고서 생성
    if z_score > 2:
        report = generate_ai_report(history[-10:])
        print(f"\n[AI 보고서]\n{report}\n")

    time.sleep(1)
```

### 프로젝트 구성

```
[미니 스마트팩토리 시스템]

  [AHT20 센서] ──→ 온습도 수집
  [스위치 SW1] ──→ 시작/정지 제어
       │
       ▼
  [Raspberry Pi] ──→ AI 이상탐지 (Z-score/이동평균)
       │              └→ 이상 시 Claude API 보고서
       │
       ├──→ [신호등 LED] R/Y/B 상태 표시
       ├──→ [WS2812 x4] RGB 세부 상태
       ├──→ [부저/스피커] 경고음
       ├──→ [OLED] 실시간 수치 표시
       └──→ [웹 대시보드] 실시간 그래프 + AI 보고서
```
