## 개요

자동차 이미지에 한국 번호판을 합성하여 AI 학습/테스트용 데이터셋을 생성합니다.

```
[자동차 배경 이미지] + [번호판 이미지 생성] → [원근 변환 합성] → [최종 이미지]
```

### 2.2 번호판 생성 프로그램

#### 파일: `scripts/generate_plate.py`

```python
"""
한국 자동차 번호판 이미지 생성 프로그램
"""

from PIL import Image, ImageDraw, ImageFont
import os

class KoreanPlateGenerator:
    """한국 번호판 이미지 생성기"""

    def __init__(self, output_dir="plates"):
        self.output_dir = output_dir
        os.makedirs(output_dir, exist_ok=True)

        # 번호판 크기 (픽셀, 실제 비율 520:110 = 4.7:1)
        self.plate_width = 520
        self.plate_height = 110

        # 폰트 설정 (시스템에 맞게 수정)
        self.font_path_korean = "C:/Windows/Fonts/malgun.ttf"  # 맑은 고딕
        self.font_path_number = "C:/Windows/Fonts/arial.ttf"   # Arial

    def generate_new_style(self, plate_number, plate_type="private"):
        """
        신형 번호판 생성 (2019.9~)

        Args:
            plate_number: 번호판 번호 (예: "123가4567")
            plate_type: "private" (비사업용), "commercial" (사업용)

        Returns:
            PIL.Image: 번호판 이미지
        """
        # 배경 색상
        if plate_type == "private":
            bg_color = (255, 255, 255)  # 흰색
            text_color = (0, 0, 0)       # 검정
        else:  # commercial
            bg_color = (255, 200, 0)     # 노랑
            text_color = (0, 0, 0)       # 검정

        # 이미지 생성
        img = Image.new('RGB', (self.plate_width, self.plate_height), bg_color)
        draw = ImageDraw.Draw(img)

        # 테두리
        draw.rectangle(
            [(2, 2), (self.plate_width-3, self.plate_height-3)],
            outline=(0, 0, 0),
            width=2
        )

        # 폰트 크기
        font_size = 70

        try:
            font = ImageFont.truetype(self.font_path_korean, font_size)
        except:
            font = ImageFont.load_default()

        # 텍스트 중앙 배치
        bbox = draw.textbbox((0, 0), plate_number, font=font)
        text_width = bbox[2] - bbox[0]
        text_height = bbox[3] - bbox[1]

        x = (self.plate_width - text_width) // 2
        y = (self.plate_height - text_height) // 2 - 5

        draw.text((x, y), plate_number, fill=text_color, font=font)

        return img

    def generate_old_style(self, plate_number):
        """
        구형 번호판 생성 (~2019.8)

        Args:
            plate_number: 번호판 번호 (예: "12가3456")

        Returns:
            PIL.Image: 번호판 이미지
        """
        # 구형은 녹색 배경도 있었으나, 여기서는 흰색으로 통일
        bg_color = (255, 255, 255)
        text_color = (0, 0, 0)

        img = Image.new('RGB', (self.plate_width, self.plate_height), bg_color)
        draw = ImageDraw.Draw(img)

        # 테두리
        draw.rectangle(
            [(2, 2), (self.plate_width-3, self.plate_height-3)],
            outline=(0, 0, 0),
            width=2
        )

        font_size = 70

        try:
            font = ImageFont.truetype(self.font_path_korean, font_size)
        except:
            font = ImageFont.load_default()

        bbox = draw.textbbox((0, 0), plate_number, font=font)
        text_width = bbox[2] - bbox[0]
        text_height = bbox[3] - bbox[1]

        x = (self.plate_width - text_width) // 2
        y = (self.plate_height - text_height) // 2 - 5

        draw.text((x, y), plate_number, fill=text_color, font=font)

        return img

    def save_plate(self, img, filename):
        """번호판 이미지 저장"""
        filepath = os.path.join(self.output_dir, filename)
        img.save(filepath, "PNG")
        return filepath

    def batch_generate(self, plate_list_file, output_dir=None):
        """
        번호판 리스트 파일로부터 대량 생성

        Args:
            plate_list_file: 번호판 목록 파일 경로
            output_dir: 출력 디렉토리
        """
        if output_dir:
            self.output_dir = output_dir
            os.makedirs(output_dir, exist_ok=True)

        with open(plate_list_file, 'r', encoding='utf-8') as f:
            plates = [line.strip() for line in f if line.strip()]

        generated = []
        for i, plate in enumerate(plates):
            # 신형/구형 판별 (앞자리 숫자 개수로)
            if len(plate) == 8:  # 신형: 123가4567
                img = self.generate_new_style(plate)
            else:  # 구형: 12가3456
                img = self.generate_old_style(plate)

            filename = f"plate_{i+1:03d}_{plate}.png"
            filepath = self.save_plate(img, filename)
            generated.append(filepath)
            print(f"생성: {filename}")

        print(f"\n총 {len(generated)}개 번호판 이미지 생성 완료")
        return generated


# 사용 예시
if __name__ == "__main__":
    generator = KoreanPlateGenerator(output_dir="../plates")

    # 단일 생성
    img = generator.generate_new_style("123가4567")
    generator.save_plate(img, "test_plate.png")

    # 대량 생성
    # generator.batch_generate("../plate_numbers.txt")
```

## 3단계: 합성

### 3.1 합성 원리

```
원본 이미지의 번호판 영역 4개 꼭지점 좌표를 지정
→ 생성된 번호판 이미지를 해당 영역에 맞게 원근 변환
→ 자연스럽게 블렌딩
```

### 3.2 좌표 지정 도구

#### 파일: `scripts/select_corners.py`

```python
"""
마우스로 번호판 영역 4개 꼭지점을 선택하는 도구
"""

import cv2
import json
import os

class CornerSelector:
    """번호판 영역 꼭지점 선택 도구"""

    def __init__(self):
        self.points = []
        self.image = None
        self.window_name = "Select 4 Corners (좌상→우상→우하→좌하)"

    def mouse_callback(self, event, x, y, flags, param):
        if event == cv2.EVENT_LBUTTONDOWN:
            if len(self.points) < 4:
                self.points.append((x, y))
                # 점 표시
                cv2.circle(self.image, (x, y), 5, (0, 255, 0), -1)
                # 점 번호 표시
                cv2.putText(self.image, str(len(self.points)),
                           (x+10, y), cv2.FONT_HERSHEY_SIMPLEX,
                           0.7, (0, 255, 0), 2)
                cv2.imshow(self.window_name, self.image)

                if len(self.points) == 4:
                    print(f"선택된 좌표: {self.points}")
                    print("Enter: 저장, R: 리셋, Q: 취소")

    def select(self, image_path):
        """
        이미지에서 4개 꼭지점 선택

        Args:
            image_path: 이미지 파일 경로

        Returns:
            list: [(x1,y1), (x2,y2), (x3,y3), (x4,y4)] 또는 None
        """
        self.image = cv2.imread(image_path)
        if self.image is None:
            print(f"이미지를 읽을 수 없습니다: {image_path}")
            return None

        self.points = []
        original = self.image.copy()

        cv2.namedWindow(self.window_name)
        cv2.setMouseCallback(self.window_name, self.mouse_callback)

        print("번호판 영역의 4개 꼭지점을 순서대로 클릭하세요:")
        print("1. 좌상단 → 2. 우상단 → 3. 우하단 → 4. 좌하단")

        cv2.imshow(self.window_name, self.image)

        while True:
            key = cv2.waitKey(1) & 0xFF

            if key == ord('r'):  # 리셋
                self.points = []
                self.image = original.copy()
                cv2.imshow(self.window_name, self.image)
                print("리셋됨. 다시 선택하세요.")

            elif key == 13:  # Enter
                if len(self.points) == 4:
                    cv2.destroyAllWindows()
                    return self.points
                else:
                    print(f"4개 점을 선택해야 합니다. 현재: {len(self.points)}개")

            elif key == ord('q'):  # 취소
                cv2.destroyAllWindows()
                return None

    def batch_select(self, image_dir, output_json):
        """
        여러 이미지의 좌표를 선택하고 JSON으로 저장

        Args:
            image_dir: 이미지 디렉토리
            output_json: 출력 JSON 파일 경로
        """
        coordinates = {}

        images = [f for f in os.listdir(image_dir)
                  if f.lower().endswith(('.jpg', '.jpeg', '.png'))]

        for img_name in images:
            img_path = os.path.join(image_dir, img_name)
            print(f"\n=== {img_name} ===")

            points = self.select(img_path)
            if points:
                coordinates[img_name] = points
                print(f"저장됨: {points}")
            else:
                print("건너뜀")

        # JSON 저장
        with open(output_json, 'w', encoding='utf-8') as f:
            json.dump(coordinates, f, indent=2, ensure_ascii=False)

        print(f"\n좌표 저장 완료: {output_json}")
        return coordinates


# 사용 예시
```

### 3.3 합성 프로그램

#### 파일: `scripts/composite_plate.py`

```python
"""
번호판 이미지를 자동차 이미지에 합성하는 프로그램
"""

import cv2
import numpy as np
import json
import os

class PlateCompositor:
    """번호판 합성기"""

    def __init__(self):
        pass

    def composite(self, background_path, plate_path, corners, output_path):
        """
        번호판을 배경 이미지에 합성

        Args:
            background_path: 배경 이미지 (자동차) 경로
            plate_path: 번호판 이미지 경로
            corners: 4개 꼭지점 좌표 [(x1,y1), (x2,y2), (x3,y3), (x4,y4)]
                     순서: 좌상 → 우상 → 우하 → 좌하
            output_path: 출력 이미지 경로

        Returns:
            bool: 성공 여부
        """
        # 이미지 로드
        background = cv2.imread(background_path)
        plate = cv2.imread(plate_path)

        if background is None or plate is None:
            print("이미지 로드 실패")
            return False

        # 번호판 이미지 크기
        h, w = plate.shape[:2]

        # 원본 번호판의 4개 꼭지점 (좌상, 우상, 우하, 좌하)
        src_points = np.float32([
            [0, 0],
            [w, 0],
            [w, h],
            [0, h]
        ])

        # 목표 위치 (배경 이미지에서의 좌표)
        dst_points = np.float32(corners)

        # 원근 변환 행렬 계산
        matrix = cv2.getPerspectiveTransform(src_points, dst_points)

        # 번호판 원근 변환
        warped_plate = cv2.warpPerspective(
            plate, matrix,
            (background.shape[1], background.shape[0])
        )

        # 마스크 생성 (번호판 영역)
        mask = np.zeros((background.shape[0], background.shape[1]), dtype=np.uint8)
        cv2.fillConvexPoly(mask, np.int32(dst_points), 255)

        # 마스크 약간 축소 (경계 부드럽게)
        kernel = np.ones((3, 3), np.uint8)
        mask = cv2.erode(mask, kernel, iterations=1)

        # 합성
        mask_inv = cv2.bitwise_not(mask)
        background_masked = cv2.bitwise_and(background, background, mask=mask_inv)
        plate_masked = cv2.bitwise_and(warped_plate, warped_plate, mask=mask)
        result = cv2.add(background_masked, plate_masked)

        # 저장
        cv2.imwrite(output_path, result)
        print(f"합성 완료: {output_path}")

        return True

    def batch_composite(self, backgrounds_dir, plates_dir, coordinates_json, output_dir):
        """
        대량 합성

        Args:
            backgrounds_dir: 배경 이미지 디렉토리
            plates_dir: 번호판 이미지 디렉토리
            coordinates_json: 좌표 JSON 파일
            output_dir: 출력 디렉토리
        """
        os.makedirs(output_dir, exist_ok=True)

        # 좌표 로드
        with open(coordinates_json, 'r', encoding='utf-8') as f:
            coordinates = json.load(f)

        # 번호판 이미지 목록
        plates = [f for f in os.listdir(plates_dir) if f.endswith('.png')]

        count = 0
        for bg_name, corners in coordinates.items():
            bg_path = os.path.join(backgrounds_dir, bg_name)

            if not os.path.exists(bg_path):
                continue

            for plate_name in plates:
                plate_path = os.path.join(plates_dir, plate_name)

                # 출력 파일명
                output_name = f"{os.path.splitext(bg_name)[0]}_{plate_name}"
                output_path = os.path.join(output_dir, output_name)

                self.composite(bg_path, plate_path, corners, output_path)
                count += 1

        print(f"\n총 {count}개 이미지 합성 완료")


# 사용 예시
if __name__ == "__main__":
    compositor = PlateCompositor()

    # 단일 합성
```

| 증강 | 효과 | 용도 |
|------|------|------|
| 노이즈 추가 | 저품질 카메라 시뮬레이션 | 실환경 대응력 |
| 밝기 조절 | 다양한 조명 조건 | 주간/야간 대응 |
| 흐림 (Blur) | 움직임, 초점 이탈 | 주행 중 촬영 시뮬레이션 |
| 회전 | 기울어진 번호판 | 다양한 각도 대응 |

### 4.2 증강 프로그램

#### 파일: `scripts/augment_images.py`

```python
"""
이미지 데이터 증강 프로그램
"""

import cv2
import numpy as np
import os
import random

class ImageAugmentor:
    """이미지 증강기"""

    def add_noise(self, image, intensity=25):
        """가우시안 노이즈 추가"""
        noise = np.random.normal(0, intensity, image.shape).astype(np.uint8)
        noisy = cv2.add(image, noise)
        return noisy

    def adjust_brightness(self, image, factor):
        """
        밝기 조절
        factor: 0.5 (어둡게) ~ 2.0 (밝게)
        """
        hsv = cv2.cvtColor(image, cv2.COLOR_BGR2HSV)
        hsv[:, :, 2] = np.clip(hsv[:, :, 2] * factor, 0, 255).astype(np.uint8)
        return cv2.cvtColor(hsv, cv2.COLOR_HSV2BGR)

    def add_blur(self, image, kernel_size=5):
        """블러 효과"""
        return cv2.GaussianBlur(image, (kernel_size, kernel_size), 0)

    def add_motion_blur(self, image, size=15):
        """모션 블러 (움직임 효과)"""
        kernel = np.zeros((size, size))
        kernel[int((size-1)/2), :] = np.ones(size)
        kernel = kernel / size
        return cv2.filter2D(image, -1, kernel)

    def augment_batch(self, input_dir, output_dir, augmentations=['noise', 'brightness', 'blur']):
        """
        디렉토리 내 모든 이미지 증강

        Args:
            input_dir: 입력 이미지 디렉토리
            output_dir: 출력 디렉토리
            augmentations: 적용할 증강 목록
        """
        os.makedirs(output_dir, exist_ok=True)

        images = [f for f in os.listdir(input_dir)
                  if f.lower().endswith(('.jpg', '.jpeg', '.png'))]

        for img_name in images:
            img_path = os.path.join(input_dir, img_name)
            image = cv2.imread(img_path)

            if image is None:
                continue

            base_name = os.path.splitext(img_name)[0]
            ext = os.path.splitext(img_name)[1]

            # 원본 복사
            cv2.imwrite(os.path.join(output_dir, img_name), image)

            # 각 증강 적용
            if 'noise' in augmentations:
                noisy = self.add_noise(image)
                cv2.imwrite(os.path.join(output_dir, f"{base_name}_noise{ext}"), noisy)

            if 'brightness' in augmentations:
                dark = self.adjust_brightness(image, 0.6)
                bright = self.adjust_brightness(image, 1.4)
                cv2.imwrite(os.path.join(output_dir, f"{base_name}_dark{ext}"), dark)
                cv2.imwrite(os.path.join(output_dir, f"{base_name}_bright{ext}"), bright)

            if 'blur' in augmentations:
                blurred = self.add_blur(image)
                cv2.imwrite(os.path.join(output_dir, f"{base_name}_blur{ext}"), blurred)

            print(f"증강 완료: {img_name}")

        print(f"\n증강 완료. 출력: {output_dir}")


# 사용 예시
```

```python
"""
합성 이미지에 대한 라벨(좌표, 번호판 텍스트) 생성
"""

import json
import os

def generate_yolo_labels(coordinates_json, plates_dir, output_dir, img_width, img_height):
    """
    YOLO 형식 라벨 생성

    Args:
        coordinates_json: 좌표 JSON 파일
        plates_dir: 번호판 디렉토리 (텍스트 추출용)
        output_dir: 라벨 출력 디렉토리
        img_width: 이미지 너비
        img_height: 이미지 높이
    """
    os.makedirs(output_dir, exist_ok=True)

    with open(coordinates_json, 'r') as f:
        coordinates = json.load(f)

    plates = [f for f in os.listdir(plates_dir) if f.endswith('.png')]

    for bg_name, corners in coordinates.items():
        # 바운딩 박스 계산
        xs = [c[0] for c in corners]
        ys = [c[1] for c in corners]

        x_min, x_max = min(xs), max(xs)
        y_min, y_max = min(ys), max(ys)

        # YOLO 형식 (정규화)
        x_center = ((x_min + x_max) / 2) / img_width
        y_center = ((y_min + y_max) / 2) / img_height
        width = (x_max - x_min) / img_width
        height = (y_max - y_min) / img_height

        for plate_name in plates:
            # 출력 파일명
            output_name = f"{os.path.splitext(bg_name)[0]}_{os.path.splitext(plate_name)[0]}.txt"
            output_path = os.path.join(output_dir, output_name)

            # YOLO 형식: class x_center y_center width height
            with open(output_path, 'w') as f:
                f.write(f"0 {x_center:.6f} {y_center:.6f} {width:.6f} {height:.6f}\n")

    print(f"라벨 생성 완료: {output_dir}")


def generate_plate_text_mapping(plates_dir, output_json):
    """
    번호판 텍스트 매핑 파일 생성

    plates_dir의 파일명에서 번호판 텍스트 추출
    예: plate_001_123가4567.png → 123가4567
    """
    mapping = {}

    for f in os.listdir(plates_dir):
        if f.endswith('.png'):
            # 파일명에서 번호판 텍스트 추출
            parts = f.replace('.png', '').split('_')
            if len(parts) >= 3:
                plate_text = parts[-1]  # 마지막 부분이 번호판 텍스트
                mapping[f] = plate_text

    with open(output_json, 'w', encoding='utf-8') as f:
        json.dump(mapping, f, indent=2, ensure_ascii=False)

    print(f"번호판 텍스트 매핑 저장: {output_json}")
    return mapping
```
