#### Three.js 3D 뷰 구현

```javascript
// 3D 모니터링 뷰 컴포넌트
class Equipment3DView {
  constructor(container) {
    this.scene = new THREE.Scene();
    this.camera = new THREE.PerspectiveCamera(60, aspectRatio, 0.1, 1000);
    this.renderer = new THREE.WebGLRenderer({ antialias: true });

    // 장비 모델 로드
    this.loader = new THREE.GLTFLoader();
    this.equipment = {};

    this.init();
  }

  async loadEquipment() {
    // 파쇄기 모델
    this.equipment.shredder = await this.loader.loadAsync('/models/shredder.glb');
    this.equipment.shredder.position.set(0, 0, 0);

    // 분쇄기 모델
    this.equipment.crusher = await this.loader.loadAsync('/models/crusher.glb');
    this.equipment.crusher.position.set(-5, 0, 3);

    // 핀밀 모델
    this.equipment.pinmill = await this.loader.loadAsync('/models/pinmill.glb');
    this.equipment.pinmill.position.set(5, 0, 3);

    // 센서 위치 마커
    this.addSensorMarkers();
  }

  addSensorMarkers() {
    // 진동 센서 (파란색)
    const vibrationGeometry = new THREE.SphereGeometry(0.1, 16, 16);
    const vibrationMaterial = new THREE.MeshBasicMaterial({ color: 0x2196F3 });

    this.sensors = {
      VIB001: this.createMarker(vibrationGeometry, vibrationMaterial, { x: 0.5, y: 0.3, z: 0 }),
      VIB002: this.createMarker(vibrationGeometry, vibrationMaterial, { x: -0.5, y: 0.3, z: 0 }),
      // ...
    };
  }

  // 실시간 상태 업데이트
  updateStatus(equipmentId, status) {
    const mesh = this.equipment[equipmentId];

    // 상태에 따른 색상 변경
    switch (status.state) {
      case 'NORMAL':
        mesh.material.emissive.setHex(0x4CAF50);
        break;
      case 'WARNING':
        mesh.material.emissive.setHex(0xFFC107);
        this.pulseEffect(mesh);
        break;
      case 'CRITICAL':
        mesh.material.emissive.setHex(0xF44336);
        this.shakeEffect(mesh);
        break;
    }

    // 회전 애니메이션 (RPM 반영)
    mesh.rotation.z += (status.rpm / 60) * 0.1;
  }
}
```

// 차트 데이터 샘플링
function sampleData(data, maxPoints = 500) {
  if (data.length <= maxPoints) return data;

  const step = Math.ceil(data.length / maxPoints);
  return data.filter((_, index) => index % step === 0);
}
