### 2.2 데이터 전처리 (Edge)

```c
// edge_preprocessing.c

typedef struct {
    float raw_value;        // 원시 값
    float filtered_value;   // 필터링된 값
    float rms_value;        // RMS 값
    float peak_value;       // 최대값
    uint32_t timestamp;     // 타임스탬프
} ProcessedSensorData_t;

// 이동평균 필터
float moving_average_filter(float* buffer, int size) {
    float sum = 0;
    for (int i = 0; i < size; i++) {
        sum += buffer[i];
    }
    return sum / size;
}

// RMS 계산
float calculate_rms(float* buffer, int size) {
    float sum_sq = 0;
    for (int i = 0; i < size; i++) {
        sum_sq += buffer[i] * buffer[i];
    }
    return sqrt(sum_sq / size);
}

// 진동 데이터 처리
ProcessedSensorData_t process_vibration(float* raw_buffer, int size) {
    ProcessedSensorData_t result;

    // DC 오프셋 제거
    float mean = calculate_mean(raw_buffer, size);
    for (int i = 0; i < size; i++) {
        raw_buffer[i] -= mean;
    }

    // 저역통과 필터 (잡음 제거)
    result.filtered_value = low_pass_filter(raw_buffer, size, 1000);  // 1kHz cutoff

    // RMS 계산
    result.rms_value = calculate_rms(raw_buffer, size);

    // Peak 값
    result.peak_value = find_max(raw_buffer, size);

    result.timestamp = HAL_GetTick();

    return result;
}

// 정규화 (0~1 범위)
float normalize_value(float value, float min, float max) {
    return (value - min) / (max - min);
}
```

### 3.1 MQTT 토픽 구조

```

├── {equipment_id}/

│   ├── status          # 상태 변경
│   ├── alert           # 알림 이벤트

├── {equipment_id}/

├── gateway/
│   ├── {gateway_id}/

  sensor/SHREDDER_01/data
```

## 5. 실시간 스트리밍

### 5.1 WebSocket 데이터 스트림

```javascript
// websocket/wsHandler.js

class RealtimeStreamHandler {
    constructor(wss, mqttClient) {
        this.wss = wss;
        this.mqttClient = mqttClient;
        this.subscriptions = new Map();  // client -> equipment_ids
    }

    handleConnection(ws, req) {
        const clientId = uuidv4();
        console.log(`Client connected: ${clientId}`);

        ws.on('message', (message) => {
            const data = JSON.parse(message);
            this.handleMessage(clientId, ws, data);
        });

        ws.on('close', () => {
            this.subscriptions.delete(clientId);
            console.log(`Client disconnected: ${clientId}`);
        });
    }

    handleMessage(clientId, ws, data) {
        switch (data.type) {
            case 'subscribe':
                this.subscribe(clientId, data.equipment_ids);
                break;

            case 'unsubscribe':
                this.unsubscribe(clientId, data.equipment_ids);
                break;

            case 'command':
                this.sendCommand(data.equipment_id, data.command);
                break;
        }
    }

    subscribe(clientId, equipmentIds) {
        this.subscriptions.set(clientId, new Set(equipmentIds));
    }

    // MQTT 메시지 수신 시 관련 클라이언트에 전달
    broadcastSensorData(equipmentId, sensorData) {
        this.wss.clients.forEach((client) => {
            const clientId = this.getClientId(client);
            const subscribedEquipment = this.subscriptions.get(clientId);

            if (subscribedEquipment?.has(equipmentId)) {
                client.send(JSON.stringify({
                    type: 'sensorUpdate',
                    equipment_id: equipmentId,
                    timestamp: Date.now(),
                    data: sensorData
                }));
            }
        });
    }
}
```

### 5.2 데이터 압축 및 최적화

```javascript
// 델타 인코딩: 변화가 있는 값만 전송
class DeltaEncoder {
    constructor() {
        this.lastValues = {};
        this.threshold = 0.01;  // 1% 변화 임계값
    }

    encode(equipmentId, newData) {
        const last = this.lastValues[equipmentId] || {};
        const delta = {};
        let hasChanges = false;

        for (const [key, value] of Object.entries(newData)) {
            const lastValue = last[key] || 0;
            const change = Math.abs(value - lastValue) / (lastValue || 1);

            if (change > this.threshold) {
                delta[key] = value;
                hasChanges = true;
            }
        }

        if (hasChanges) {
            this.lastValues[equipmentId] = { ...last, ...delta };
            return delta;
        }

        return null;  // 변화 없음
    }
}

// 배치 전송: 여러 업데이트를 묶어서 전송
class BatchSender {
    constructor(ws, interval = 100) {
        this.ws = ws;
        this.buffer = [];
        this.interval = interval;

        setInterval(() => this.flush(), interval);
    }

    add(data) {
        this.buffer.push(data);
    }

    flush() {
        if (this.buffer.length > 0) {
            this.ws.send(JSON.stringify({
                type: 'batch',
                items: this.buffer,
                timestamp: Date.now()
            }));
            this.buffer = [];
        }
    }
}
```
