#include "tensorflow/lite/micro/micro_interpreter.h"

// TFLite 모델 데이터 (Flash에 저장)

extern const int anomaly_model_tflite_len;

static tflite::MicroInterpreter* interpreter;

static TfLiteTensor* output;

void AI_Init(void) {
    // 모델 로드 및 인터프리터 초기화
    static tflite::AllOpsResolver resolver;
    static uint8_t tensor_arena[50 * 1024];  // 50KB

    interpreter = new tflite::MicroInterpreter(
        model, resolver, tensor_arena, sizeof(tensor_arena)
    );
    interpreter->AllocateTensors();

    input = interpreter->input(0);
    output = interpreter->output(0);
}

float AI_RunInference(float* sensor_data, int len) {
    // 입력 데이터 복사
    for (int i = 0; i < len; i++) {
        input->data.f[i] = sensor_data[i];
    }

    // 추론 실행
    interpreter->Invoke();

    // 결과 반환 (Anomaly Score)
    return output->data.f[0];
}

// control_logic.c - 과부하 제어 로직
typedef enum {
    STATE_NORMAL,
    STATE_WARNING,
    STATE_OVERLOAD_RETRY,
    STATE_EMERGENCY_STOP
} ControlState_t;

static ControlState_t currentState = STATE_NORMAL;
static int retryCount = 0;

void Control_Process(float anomalyScore, float current) {
    switch (currentState) {
        case STATE_NORMAL:
            if (anomalyScore > 0.8 || current > 150) {
                currentState = STATE_WARNING;
                Alert_Send(ALERT_WARNING);
            }
            break;

        case STATE_WARNING:
            if (anomalyScore > 0.9 || current > 180) {
                currentState = STATE_OVERLOAD_RETRY;
                Inverter_ReduceSpeed(50);  // 50% 감속
            } else if (anomalyScore < 0.5 && current < 120) {
                currentState = STATE_NORMAL;
            }
            break;

        case STATE_OVERLOAD_RETRY:
            if (current > 150) {
                retryCount++;
                Inverter_Reverse();  // 역회전
                vTaskDelay(2000);    // 2초 대기
                Inverter_Forward();  // 정회전

                if (retryCount >= 3) {
                    currentState = STATE_EMERGENCY_STOP;
                    Inverter_Stop();
                    Alert_Send(ALERT_EMERGENCY);
                }
            } else {
                currentState = STATE_WARNING;
                retryCount = 0;
            }
            break;

        case STATE_EMERGENCY_STOP:
            // 수동 리셋 대기
            break;
    }
}

// services/mqttService.js
const mqtt = require('mqtt');
const { InfluxDB, Point } = require('@influxdata/influxdb-client');

class MQTTService {
    constructor() {
        this.client = mqtt.connect(process.env.MQTT_BROKER_URL);
        this.influx = new InfluxDB({
            url: process.env.INFLUXDB_URL,
            token: process.env.INFLUXDB_TOKEN
        });
        this.writeApi = this.influx.getWriteApi('uttec', 'sensor_data');
    }

    init() {
        this.client.on('connect', () => {
            console.log('MQTT Connected');
            this.client.subscribe('sensor/+/data');
            this.client.subscribe('sensor/+/alert');
        });

        this.client.on('message', (topic, message) => {
            this.handleMessage(topic, JSON.parse(message.toString()));
        });
    }

    handleMessage(topic, data) {
        const [, equipmentId, type] = topic.split('/');

        if (type === 'data') {
            // 센서 데이터 저장
            const point = new Point('sensor')
                .tag('equipment', equipmentId)
                .floatField('vibration_de', data.vib1)
                .floatField('vibration_nde', data.vib2)
                .floatField('temperature_motor', data.temp1)
                .floatField('current_motor1', data.curr1)
                .floatField('speed', data.speed)
                .floatField('anomaly_score', data.anomalyScore)
                .timestamp(new Date());

            this.writeApi.writePoint(point);

            // WebSocket으로 실시간 전송
            this.broadcast('sensorUpdate', { equipmentId, data });
        }

        if (type === 'alert') {
            // 알림 처리
            this.alertService.send(data);
        }
    }
}

module.exports = new MQTTService();

// hooks/useWebSocket.js
import { useEffect, useRef, useCallback } from 'react';
import { useDispatch } from 'react-redux';
import { updateSensorData } from '../store/sensorSlice';

export const useWebSocket = (url) => {
    const ws = useRef(null);
    const dispatch = useDispatch();

    const connect = useCallback(() => {
        ws.current = new WebSocket(url);

        ws.current.onopen = () => {
            console.log('WebSocket connected');
        };

        ws.current.onmessage = (event) => {
            const data = JSON.parse(event.data);

            switch(data.type) {
                case 'sensorUpdate':
                    dispatch(updateSensorData(data.payload));
                    break;
                case 'alert':
                    // 알림 처리
                    break;
            }
        };

        ws.current.onclose = () => {
            console.log('WebSocket disconnected, reconnecting...');
            setTimeout(connect, 3000);
        };
    }, [url, dispatch]);

    useEffect(() => {
        connect();
        return () => ws.current?.close();
    }, [connect]);

    const send = (message) => {
        if (ws.current?.readyState === WebSocket.OPEN) {
            ws.current.send(JSON.stringify(message));
        }
    };

    return { send };
};
