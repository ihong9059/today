```c
// main.c - 메인 엔트리
int main(void) {
    HAL_Init();
    SystemClock_Config();

    // 드라이버 초기화
    ADC_Init();
    UART_Init();
    ETH_Init();

    // 미들웨어 초기화
    TFLite_Init();
    Modbus_Init();
    MQTT_Init();

    // 태스크 생성
    xTaskCreate(SensorTask, "Sensor", 512, NULL, 3, NULL);
    xTaskCreate(AITask, "AI", 1024, NULL, 2, NULL);
    xTaskCreate(ControlTask, "Control", 512, NULL, 3, NULL);
    xTaskCreate(CommTask, "Comm", 512, NULL, 2, NULL);
    xTaskCreate(DiagTask, "Diag", 256, NULL, 1, NULL);

    // 스케줄러 시작
    vTaskStartScheduler();

    while(1);
}

// sensor_task.c - 센서 태스크
void SensorTask(void *pvParameters) {
    TickType_t xLastWakeTime = xTaskGetTickCount();

    while(1) {
        // ADC 샘플링
        ReadAllSensors(&sensorData);

        // 신호 처리
        ProcessVibration(&sensorData.vibration);
        ProcessTemperature(&sensorData.temperature);
        ProcessTension(&sensorData.tension);

        // 특징 추출
        ExtractFeatures(&sensorData, &features);

        // 큐에 전송
        xQueueSend(aiQueue, &features, 0);
        xQueueSend(commQueue, &sensorData, 0);

        vTaskDelayUntil(&xLastWakeTime, pdMS_TO_TICKS(10));
    }
}

// ai_task.c - AI 추론 태스크
void AITask(void *pvParameters) {
    Features_t features;
    AIResult_t result;

    while(1) {
        if(xQueueReceive(aiQueue, &features, portMAX_DELAY)) {
            // 이상 탐지 추론
            result.anomalyScore = TFLite_Inference_Anomaly(&features);

            // 품질 예측
            result.qualityScore = TFLite_Inference_Quality(&features);

            // 임계값 체크
            if(result.anomalyScore > THRESHOLD_WARNING) {
                xQueueSend(alertQueue, &result, 0);
            }

            // 결과 전송
            xQueueSend(commQueue, &result, 0);
        }
    }
}
```

### 3.4 Alert Service

```javascript
// alert_service.js - 알림 서비스
const nodemailer = require('nodemailer');
const axios = require('axios');

class AlertService {
    constructor() {
        this.emailTransporter = nodemailer.createTransporter({
            service: 'gmail',
            auth: {
                user: process.env.EMAIL_USER,
                pass: process.env.EMAIL_PASS
            }
        });
    }

    async sendAlert(alertData) {
        const { severity, equipment, message } = alertData;

        // 알림 채널 선택
        if (severity === 'critical') {
            await Promise.all([
                this.sendSMS(alertData),
                this.sendEmail(alertData),
                this.sendKakao(alertData),
                this.sendPush(alertData)
            ]);
        } else if (severity === 'warning') {
            await Promise.all([
                this.sendEmail(alertData),
                this.sendPush(alertData)
            ]);
        } else {
            await this.sendPush(alertData);
        }

        // DB 저장
        await this.saveAlertLog(alertData);
    }

    async sendSMS(alertData) {
        // Twilio 또는 국내 SMS API
        const response = await axios.post(process.env.SMS_API_URL, {
            to: alertData.phoneNumbers,
            message: this.formatSMS(alertData)
        });
        return response.data;
    }

    async sendKakao(alertData) {
        // 카카오 알림톡 API
        const response = await axios.post(process.env.KAKAO_API_URL, {
            template_code: 'ALERT_001',
            receivers: alertData.kakaoIds,
            content: this.formatKakao(alertData)
        });
        return response.data;
    }

    formatSMS(alertData) {
        return `[필에너지 알림]
설비: ${alertData.equipment}
상태: ${alertData.severity}
내용: ${alertData.message}
시간: ${new Date().toLocaleString('ko-KR')}`;
    }
}

module.exports = new AlertService();
```

```javascript
// hooks/useWebSocket.js
import { useEffect, useState } from 'react';
import { io } from 'socket.io-client';

export const useWebSocket = (equipmentId) => {
    const [sensorData, setSensorData] = useState(null);
    const [aiResult, setAiResult] = useState(null);
    const [connected, setConnected] = useState(false);

    useEffect(() => {
        const socket = io(process.env.REACT_APP_WS_URL);

        socket.on('connect', () => {
            setConnected(true);
            socket.emit('subscribe', equipmentId);
        });

        socket.on('sensor-data', (data) => {
            if (data.equipmentId === equipmentId) {
                setSensorData(data);
            }
        });

        socket.on('ai-result', (data) => {
            if (data.equipmentId === equipmentId) {
                setAiResult(data);
            }
        });

        socket.on('disconnect', () => {
            setConnected(false);
        });

        return () => socket.disconnect();
    }, [equipmentId]);

    return { sensorData, aiResult, connected };
};
```

### 5.1 비전 시스템 구조

```python
# vision_system.py - 비전 AI 시스템
import cv2
import numpy as np
from ultralytics import YOLO
import paho.mqtt.client as mqtt

class VisionSystem:
    def __init__(self):
        # 모델 로드
        self.alignment_model = YOLO('models/alignment_yolo.pt')
        self.notch_model = load_model('models/notch_classifier.h5')

        # 카메라 초기화
        self.cameras = {
            'stacking': cv2.VideoCapture(0),
            'notching': cv2.VideoCapture(1),
            'welding': cv2.VideoCapture(2)
        }

        # MQTT 클라이언트
        self.mqtt_client = mqtt.Client()
        self.mqtt_client.connect('localhost', 1883)

    def process_stacking(self, frame):
        """Stacking 정렬도 검사"""
        # YOLO 추론
        results = self.alignment_model(frame)

        # 키포인트 추출
        keypoints = extract_keypoints(results)

        # 정렬 오차 계산
        alignment_error = calculate_alignment(keypoints)

        # 판정
        if alignment_error < 0.05:
            grade = 'OK'
        else:
            grade = 'NG'

        # 결과 발행
        self.mqtt_client.publish('vision/stacking', json.dumps({
            'alignmentError': alignment_error,
            'grade': grade,
            'timestamp': time.time()
        }))

        return alignment_error, grade

    def process_notching(self, frame):
        """Notching 품질 검사"""
        # ROI 추출
        roi = extract_notch_roi(frame)

        # 전처리
        processed = preprocess_notch(roi)

        # 분류
        prediction = self.notch_model.predict(processed)
        class_names = ['OK', 'Burr', 'Incomplete', 'Crack']
        result = class_names[np.argmax(prediction)]

        # 결과 발행
        self.mqtt_client.publish('vision/notching', json.dumps({
            'result': result,
            'confidence': float(np.max(prediction)),
            'timestamp': time.time()
        }))

        return result

    def run(self):
        """메인 루프"""
        while True:
            # 각 카메라에서 프레임 캡처
            for name, cap in self.cameras.items():
                ret, frame = cap.read()
                if ret:
                    if name == 'stacking':
                        self.process_stacking(frame)
                    elif name == 'notching':
                        self.process_notching(frame)

if __name__ == '__main__':
    vision = VisionSystem()
    vision.run()
```

```sql
CREATE TABLE equipment (

    name VARCHAR(100) NOT NULL,
    type VARCHAR(50) NOT NULL,
    location VARCHAR(100),
    status VARCHAR(20) DEFAULT 'normal',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE alerts (

    acknowledged BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 정비 기록 테이블

);
```
