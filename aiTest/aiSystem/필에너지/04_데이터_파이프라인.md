```c
// sensor_collector.c - Edge Gateway 센서 수집
#include "sensor_collector.h"
#include "mqtt_client.h"

// 센서 설정
SensorConfig_t sensorConfigs[] = {
    { .id = "VIB_1", .channel = ADC1_CH0, .type = VIBRATION, .sampleRate = 100 },
    { .id = "VIB_2", .channel = ADC1_CH1, .type = VIBRATION, .sampleRate = 100 },
    { .id = "TEMP_1", .channel = ADC1_CH4, .type = TEMPERATURE, .sampleRate = 1 },
    { .id = "TENS_1", .channel = ADC2_CH0, .type = TENSION, .sampleRate = 100 },
    // ... 16개 센서
};

void SensorCollectorTask(void *pvParameters) {
    SensorData_t data;
    char jsonBuffer[512];
    char topic[64];

    while (1) {
        // 모든 센서 읽기
        for (int i = 0; i < NUM_SENSORS; i++) {
            // ADC 읽기
            uint16_t rawValue = ADC_Read(sensorConfigs[i].channel);

            // 스케일 변환
            data.value = ConvertToPhysical(rawValue, sensorConfigs[i].type);
            data.timestamp = GetTimestamp();
            strcpy(data.sensorId, sensorConfigs[i].id);

            // 신호 처리
            if (sensorConfigs[i].type == VIBRATION) {
                data.value = CalculateRMS(data.value, i);
            }

            // JSON 직렬화
            sprintf(jsonBuffer,
                "{\"sensorId\":\"%s\",\"value\":%.3f,\"timestamp\":%lu}",
                data.sensorId, data.value, data.timestamp);

            // MQTT 발행
            sprintf(topic, "sensor/%s/%s", EQUIPMENT_ID, data.sensorId);
            MQTT_Publish(topic, jsonBuffer, QOS_0);
        }

        vTaskDelay(pdMS_TO_TICKS(10)); // 10ms 주기
    }
}

// RMS 계산 (슬라이딩 윈도우)
float CalculateRMS(float newSample, int sensorIndex) {
    static float buffer[NUM_SENSORS][RMS_WINDOW_SIZE];
    static int index[NUM_SENSORS] = {0};

    buffer[sensorIndex][index[sensorIndex]] = newSample * newSample;
    index[sensorIndex] = (index[sensorIndex] + 1) % RMS_WINDOW_SIZE;

    float sum = 0;
    for (int i = 0; i < RMS_WINDOW_SIZE; i++) {
        sum += buffer[sensorIndex][i];
    }

    return sqrt(sum / RMS_WINDOW_SIZE);
}
```

## 4. 데이터 수집 서비스

### 4.1 Data Ingester

```python
# data_ingester.py - 데이터 수집 서비스
import paho.mqtt.client as mqtt
from influxdb_client import InfluxDBClient, Point
from influxdb_client.client.write_api import SYNCHRONOUS
import psycopg2
import json
import logging

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class DataIngester:
    def __init__(self, config):
        # MQTT 클라이언트
        self.mqtt_client = mqtt.Client()
        self.mqtt_client.on_connect = self.on_connect
        self.mqtt_client.on_message = self.on_message

        # InfluxDB 클라이언트
        self.influx_client = InfluxDBClient(
            url=config['influxdb_url'],
            token=config['influxdb_token'],
            org=config['influxdb_org']
        )
        self.write_api = self.influx_client.write_api(write_options=SYNCHRONOUS)

        # PostgreSQL 연결
        self.pg_conn = psycopg2.connect(config['postgres_url'])

        # 배치 버퍼
        self.batch_buffer = []
        self.batch_size = 100

    def on_connect(self, client, userdata, flags, rc):
        logger.info("Connected to MQTT broker")
        # 토픽 구독
        client.subscribe("philenergy/sensor/#")
        client.subscribe("philenergy/ai/#")
        client.subscribe("philenergy/vision/#")
        client.subscribe("philenergy/alert/#")

    def on_message(self, client, userdata, msg):
        try:
            topic = msg.topic
            payload = json.loads(msg.payload.decode())

            # 토픽 파싱
            parts = topic.split('/')
            data_type = parts[1]  # sensor, ai, vision, alert
            equipment_id = parts[2]

            if data_type == 'sensor':
                self.process_sensor_data(equipment_id, parts[3], payload)
            elif data_type == 'ai':
                self.process_ai_result(equipment_id, parts[3], payload)
            elif data_type == 'vision':
                self.process_vision_result(equipment_id, parts[3], payload)
            elif data_type == 'alert':
                self.process_alert(equipment_id, payload)

        except Exception as e:
            logger.error(f"Error processing message: {e}")

    def process_sensor_data(self, equipment_id, sensor_type, data):
        """센서 데이터 처리 → InfluxDB"""
        point = Point("sensor_data") \
            .tag("equipment_id", equipment_id) \
            .tag("sensor_type", sensor_type) \
            .tag("sensor_id", data['sensorId']) \
            .field("value", float(data['value'])) \
            .time(data['timestamp'])

        self.batch_buffer.append(point)

        # 배치 쓰기
        if len(self.batch_buffer) >= self.batch_size:
            self.write_api.write(
                bucket="philenergy",
                record=self.batch_buffer
            )
            self.batch_buffer = []
            logger.debug(f"Wrote {self.batch_size} points to InfluxDB")

    def process_ai_result(self, equipment_id, model_type, data):
        """AI 결과 처리 → InfluxDB"""
        point = Point("ai_result") \
            .tag("equipment_id", equipment_id) \
            .tag("model_type", model_type) \
            .field("anomaly_score", float(data.get('anomalyScore', 0))) \
            .field("status", data.get('status', 'unknown')) \
            .time(data['timestamp'])

        self.write_api.write(bucket="philenergy", record=point)

    def process_vision_result(self, equipment_id, inspection_type, data):
        """비전 검사 결과 처리 → InfluxDB + PostgreSQL"""
        # InfluxDB (시계열)
        point = Point("vision_result") \
            .tag("equipment_id", equipment_id) \
            .tag("inspection_type", inspection_type) \
            .field("result", data['result']) \
            .field("confidence", float(data['confidence'])) \
            .time(data['timestamp'])

        self.write_api.write(bucket="philenergy", record=point)

        # PostgreSQL (상세 기록)
        cursor = self.pg_conn.cursor()
        cursor.execute("""
            INSERT INTO quality_results
            (equipment_id, inspection_type, result, confidence, details, inspected_at)
            VALUES (%s, %s, %s, %s, %s, to_timestamp(%s/1000))
        """, (
            equipment_id,
            inspection_type,
            data['result'],
            data['confidence'],
            json.dumps(data),
            data['timestamp']
        ))
        self.pg_conn.commit()

    def process_alert(self, equipment_id, data):
        """알람 처리 → PostgreSQL"""
        cursor = self.pg_conn.cursor()
        cursor.execute("""
            INSERT INTO alerts
            (equipment_id, severity, message, details, created_at)
            VALUES (%s, %s, %s, %s, to_timestamp(%s/1000))
        """, (
            equipment_id,
            data['severity'],
            data['message'],
            json.dumps(data.get('details', {})),
            data['timestamp']
        ))
        self.pg_conn.commit()

        logger.warning(f"Alert: [{data['severity']}] {equipment_id} - {data['message']}")

    def run(self):
        """서비스 시작"""
        self.mqtt_client.connect("localhost", 1883, 60)
        self.mqtt_client.loop_forever()

if __name__ == '__main__':
    config = {
        'influxdb_url': 'http://localhost:8086',
        'influxdb_token': 'your-token',
        'influxdb_org': 'philenergy',
        'postgres_url': 'postgresql://user:pass@localhost/philenergy'
    }

    ingester = DataIngester(config)
    ingester.run()
```

### 6.1 실시간 스트림 처리

```python
# stream_processor.py - 실시간 스트림 처리
import asyncio
from collections import deque
import numpy as np

class StreamProcessor:
    def __init__(self, window_size=100):
        self.window_size = window_size
        self.buffers = {}

    async def process_stream(self, equipment_id, sensor_type, value):
        """실시간 스트림 처리"""
        key = f"{equipment_id}/{sensor_type}"

        # 버퍼 초기화
        if key not in self.buffers:
            self.buffers[key] = deque(maxlen=self.window_size)

        self.buffers[key].append(value)

        # 윈도우가 차면 특징 추출
        if len(self.buffers[key]) == self.window_size:
            features = self.extract_features(self.buffers[key])
            return features

        return None

    def extract_features(self, data):
        """특징 추출"""
        arr = np.array(data)
        return {
            'mean': float(np.mean(arr)),
            'std': float(np.std(arr)),
            'max': float(np.max(arr)),
            'min': float(np.min(arr)),
            'rms': float(np.sqrt(np.mean(arr**2))),
            'kurtosis': float(self.kurtosis(arr)),
            'skewness': float(self.skewness(arr))
        }

    @staticmethod
    def kurtosis(arr):
        n = len(arr)
        mean = np.mean(arr)
        std = np.std(arr)
        if std == 0:
            return 0
        return np.sum((arr - mean)**4) / (n * std**4) - 3

    @staticmethod
    def skewness(arr):
        n = len(arr)
        mean = np.mean(arr)
        std = np.std(arr)
        if std == 0:
            return 0
        return np.sum((arr - mean)**3) / (n * std**3)
```

```python
# batch_pipeline.py - 배치 처리 파이프라인
from datetime import datetime, timedelta
import pandas as pd
from influxdb_client import InfluxDBClient

class BatchPipeline:
    def __init__(self, config):
        self.influx_client = InfluxDBClient(
            url=config['influxdb_url'],
            token=config['influxdb_token']
        )
        self.query_api = self.influx_client.query_api()

    def run_daily_analysis(self, date=None):
        """일간 분석 파이프라인"""
        if date is None:
            date = datetime.now() - timedelta(days=1)

        # 1. 데이터 수집
        sensor_data = self.fetch_sensor_data(date)
        ai_results = self.fetch_ai_results(date)
        quality_results = self.fetch_quality_results(date)

        # 2. 통계 계산
        daily_stats = self.calculate_daily_stats(sensor_data, ai_results)

        # 3. 트렌드 분석
        trends = self.analyze_trends(sensor_data)

        # 4. 이상 패턴 분석
        anomaly_patterns = self.analyze_anomaly_patterns(ai_results)

        # 5. 품질 통계
        quality_stats = self.calculate_quality_stats(quality_results)

        # 6. 리포트 생성
        report = self.generate_report(daily_stats, trends, anomaly_patterns, quality_stats)

        # 7. 저장
        self.save_report(report, date)

        return report

    def fetch_sensor_data(self, date):
        """센서 데이터 조회"""
        query = f'''
        from(bucket: "philenergy")
            |> range(start: {date.strftime('%Y-%m-%dT00:00:00Z')},
                     stop: {date.strftime('%Y-%m-%dT23:59:59Z')})
            |> filter(fn: (r) => r._measurement == "sensor_data")
            |> aggregateWindow(every: 1m, fn: mean)
        '''
        return self.query_api.query_data_frame(query)

    def calculate_daily_stats(self, sensor_data, ai_results):
        """일간 통계 계산"""
        stats = {
            'equipment_uptime': self.calculate_uptime(sensor_data),
            'avg_anomaly_score': ai_results['anomaly_score'].mean(),
            'max_anomaly_score': ai_results['anomaly_score'].max(),
            'alert_count': len(ai_results[ai_results['anomaly_score'] > 0.5]),
            'sensor_stats': {}
        }

        # 센서별 통계
        for sensor in sensor_data['sensor_id'].unique():
            sensor_df = sensor_data[sensor_data['sensor_id'] == sensor]
            stats['sensor_stats'][sensor] = {
                'mean': sensor_df['value'].mean(),
                'std': sensor_df['value'].std(),
                'max': sensor_df['value'].max(),
                'min': sensor_df['value'].min()
            }

        return stats

    def analyze_trends(self, sensor_data):
        """트렌드 분석"""
        trends = {}

        for sensor in sensor_data['sensor_id'].unique():
            sensor_df = sensor_data[sensor_data['sensor_id'] == sensor].sort_values('_time')

            # 선형 회귀로 트렌드 계산
            x = np.arange(len(sensor_df))
            y = sensor_df['value'].values

            if len(x) > 1:
                slope, _ = np.polyfit(x, y, 1)
                trends[sensor] = {
                    'slope': slope,
                    'direction': 'increasing' if slope > 0 else 'decreasing',
                    'significance': abs(slope) > 0.01
                }

        return trends

    def generate_report(self, daily_stats, trends, anomaly_patterns, quality_stats):
        """리포트 생성"""
        return {
            'date': datetime.now().isoformat(),
            'summary': {
                'overall_health': self.calculate_overall_health(daily_stats),
                'uptime_percent': daily_stats['equipment_uptime'],
                'quality_rate': quality_stats['ok_rate']
            },
            'statistics': daily_stats,
            'trends': trends,
            'anomaly_analysis': anomaly_patterns,
            'quality_analysis': quality_stats,
            'recommendations': self.generate_recommendations(daily_stats, trends)
        }
```

### 7.2 백업 스크립트

```bash
#!/bin/bash
# backup.sh - 일간 백업 스크립트

DATE=$(date +%Y%m%d)
BACKUP_DIR="/backup/$DATE"

# InfluxDB 백업
influx backup $BACKUP_DIR/influxdb \
    --bucket philenergy \
    --host http://localhost:8086

# PostgreSQL 백업
pg_dump -h localhost -U postgres philenergy > $BACKUP_DIR/postgresql.sql

# 압축 및 S3 업로드
tar -czf $BACKUP_DIR.tar.gz $BACKUP_DIR
aws s3 cp $BACKUP_DIR.tar.gz s3://philenergy-backup/

# 오래된 백업 삭제
find /backup -type d -mtime +30 -exec rm -rf {} \;

echo "Backup completed: $DATE"
```
