## 1. UI/UX 개요

### 1.1 디자인 원칙

| 원칙 | 설명 |
|------|------|
| **직관성** | 복잡한 설비 데이터를 한눈에 파악 가능 |
| **실시간성** | 센서 데이터와 알람을 즉시 표시 |
| **접근성** | 다양한 기기(PC, 태블릿, 모바일)에서 사용 |
| **일관성** | 통일된 색상과 컴포넌트 사용 |

### 4.2 센서 게이지 컴포넌트

```jsx
// SensorGauge.jsx
const SensorGauge = ({ sensor }) => {
    const getStatusColor = (value, thresholds) => {
        if (value >= thresholds.error) return '#f44336';
        if (value >= thresholds.warning) return '#ff9800';
        return '#4caf50';
    };

    const percentage = (sensor.value / sensor.max) * 100;
    const color = getStatusColor(sensor.value, sensor.thresholds);

    return (
        <div className="sensor-gauge">
            <div className="gauge-header">
                <span className="sensor-name">{sensor.name}</span>
                <span className="sensor-type">{sensor.type}</span>
            </div>
            <div className="gauge-body">
                <div className="gauge-value" style={{ color }}>
                    {sensor.value.toFixed(1)}
                </div>
                <div className="gauge-unit">{sensor.unit}</div>
                <div className="gauge-bar">
                    <div
                        className="gauge-fill"
                        style={{
                            width: `${percentage}%`,
                            backgroundColor: color
                        }}
                    />
                </div>
            </div>
            <div className="gauge-footer">
                <span>0</span>
                <span className="status-dot" style={{ backgroundColor: color }} />
                <span>{sensor.max}</span>
            </div>
        </div>
    );
};
```

```jsx
// AnomalyScore.jsx
const AnomalyScore = ({ score, status, features }) => {
    const getStatusInfo = (status) => {
        const statusMap = {
            normal: { color: '#4caf50', text: 'NORMAL', icon: '✓' },
            warning: { color: '#ff9800', text: 'WARNING', icon: '⚠' },
            error: { color: '#f44336', text: 'ERROR', icon: '✕' }
        };
        return statusMap[status];
    };

    const info = getStatusInfo(status);
    const percentage = Math.min(score * 100, 100);

    return (
        <div className="anomaly-score-card">
            <h3>Anomaly Detection</h3>

            <div className="score-display">
                <div className="score-value" style={{ color: info.color }}>
                    {score.toFixed(2)}
                </div>
                <div className="score-bar">
                    <div
                        className="score-fill"
                        style={{
                            width: `${percentage}%`,
                            backgroundColor: info.color
                        }}
                    />
                    <div className="threshold-line" style={{ left: '50%' }} />
                </div>
                <div className="status-badge" style={{ backgroundColor: info.color }}>
                    {info.icon} {info.text}
                </div>
            </div>

            <div className="feature-contributions">
                <h4>Contributing Factors</h4>
                {features.map(feature => (
                    <div key={feature.name} className="feature-item">
                        <span className="feature-name">{feature.name}</span>
                        <div className="feature-bar">
                            <div
                                className="feature-fill"
                                style={{ width: `${feature.contribution * 100}%` }}
                            />
                        </div>
                        <span className="feature-value">
                            {(feature.contribution * 100).toFixed(0)}%
                        </span>
                    </div>
                ))}
            </div>
        </div>
    );
};
```

```jsx
    };

    return (
        <div className="rul-indicator-card">

                return (
                    <div key={pred.component} className="rul-item">
                        <div className="rul-header">
                            <span className="component-name">{pred.component}</span>
                            <span className="rul-hours">{pred.rulHours.toLocaleString()}h</span>
                        </div>

                        <div className="rul-bar">
                            <div
                                className="rul-fill"
                                style={{
                                    width: `${percentage}%`,
                                    backgroundColor: color
                                }}
                            />
                        </div>

                        <div className="rul-footer">
                            <span className="percentage">{percentage.toFixed(0)}% remaining</span>
                            <span className="maintenance-date">
                                교체 예정: {pred.maintenanceDate}
                            </span>
                        </div>

                        {pred.recommendation && (
                            <div className="recommendation">
                                <span className="icon">💡</span>
                                {pred.recommendation}
                            </div>
                        )}
                    </div>
                );
            })}
        </div>
    );
};
```

```jsx
// AlertCard.jsx
const AlertCard = ({ alert, onAcknowledge, onViewDetail }) => {
    const severityConfig = {
        critical: { color: '#f44336', icon: '🔴', bg: 'rgba(244,67,54,0.1)' },
        warning: { color: '#ff9800', icon: '🟡', bg: 'rgba(255,152,0,0.1)' },
        info: { color: '#2196f3', icon: '🔵', bg: 'rgba(33,150,243,0.1)' }
    };

    const config = severityConfig[alert.severity];

    return (
        <div
            className="alert-card"
            style={{ backgroundColor: config.bg, borderLeftColor: config.color }}
        >
            <div className="alert-header">
                <span className="severity-badge">
                    {config.icon} {alert.severity.toUpperCase()}
                </span>
                <span className="alert-title">{alert.message}</span>
            </div>

            <div className="alert-meta">
                <span className="equipment">{alert.equipmentId}</span>
                <span className="timestamp">
                    {new Date(alert.createdAt).toLocaleString('ko-KR')}
                </span>
            </div>

            <div className="alert-details">
                {alert.details.sensorValue && (
                    <span>
                        {alert.details.sensor}: {alert.details.sensorValue}
                        (threshold: {alert.details.threshold})
                    </span>
                )}
            </div>

            <div className="alert-actions">
                {!alert.acknowledged && (
                    <button
                        className="btn-acknowledge"
                        onClick={() => onAcknowledge(alert.id)}
                    >
                        확인
                    </button>
                )}
                <button
                    className="btn-detail"
                    onClick={() => onViewDetail(alert.id)}
                >
                    상세보기
                </button>
            </div>

            {alert.acknowledged && (
                <div className="acknowledged-info">
                    ✓ 확인됨 by {alert.acknowledgedBy} at{' '}
                    {new Date(alert.acknowledgedAt).toLocaleString('ko-KR')}
                </div>
            )}
        </div>
    );
};
```
