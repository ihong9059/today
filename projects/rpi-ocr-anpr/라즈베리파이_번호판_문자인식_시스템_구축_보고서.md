## 1. 프로젝트 개요

### 1.1 목표
라즈베리파이 3 B+를 활용하여 실시간으로 자동차 번호판을 인식하거나 일반 문자를 인식하는 시스템을 구축합니다.

### 1.2 주요 기능
- **번호판 인식 (ANPR)**: 차량 번호판 검출 및 문자 추출
- **일반 OCR**: 문서, 표지판 등의 문자 인식
- **실시간 처리**: 카메라 스트리밍 기반 인식
- **데이터 로깅**: 인식 결과 저장 및 관리

### 1.3 라즈베리파이 3 B+ 사양

| 항목 | 사양 |
|------|------|
| CPU | Broadcom BCM2837B0, Cortex-A53 64-bit 1.4GHz (쿼드코어) |
| RAM | 1GB LPDDR2 |
| GPU | VideoCore IV 400MHz |
| 네트워크 | Gigabit Ethernet (USB 2.0 한계로 ~300Mbps), 듀얼밴드 WiFi |
| USB | 4x USB 2.0 |
| 카메라 | CSI 카메라 포트 |
| GPIO | 40핀 |
| 전원 | 5V/2.5A |

### 1.4 제약 사항
- **메모리 제한**: 1GB RAM으로 대형 모델 사용 불가
- **처리 속도**: CPU 기반 처리로 실시간 고해상도 처리 어려움
- **발열**: 장시간 고부하 시 쓰로틀링 발생 가능

### 5.2 카메라 모듈 (camera.py)

```python
# src/camera.py
import cv2
import time
import sys

class CameraModule:
    def __init__(self, config):
        self.config = config
        self.cap = None
        self.use_picamera = config.get('use_picamera', True)

    def initialize(self):
        """카메라 초기화"""
        if self.use_picamera:
            try:
                from picamera2 import Picamera2
                self.cap = Picamera2()
                camera_config = self.cap.create_preview_configuration(
                    main={"size": (self.config['width'], self.config['height'])}
                )
                self.cap.configure(camera_config)
                self.cap.start()
                time.sleep(2)  # 카메라 웜업
                print("[INFO] PiCamera2 초기화 완료")
                return True
            except Exception as e:
                print(f"[WARN] PiCamera2 초기화 실패: {e}")
                self.use_picamera = False

        # USB 웹캠 폴백
        self.cap = cv2.VideoCapture(0)
        if not self.cap.isOpened():
            print("[ERROR] 카메라를 열 수 없습니다")
            return False

        self.cap.set(cv2.CAP_PROP_FRAME_WIDTH, self.config['width'])
        self.cap.set(cv2.CAP_PROP_FRAME_HEIGHT, self.config['height'])
        self.cap.set(cv2.CAP_PROP_FPS, self.config['fps'])
        print("[INFO] USB 웹캠 초기화 완료")
        return True

    def read_frame(self):
        """프레임 읽기"""
        if self.use_picamera:
            frame = self.cap.capture_array()
            # RGB to BGR 변환 (OpenCV 호환)
            frame = cv2.cvtColor(frame, cv2.COLOR_RGB2BGR)
            return True, frame
        else:
            return self.cap.read()

    def release(self):
        """카메라 해제"""
        if self.cap:
            if self.use_picamera:
                self.cap.stop()
            else:
                self.cap.release()
        print("[INFO] 카메라 해제 완료")


# 테스트
if __name__ == "__main__":
    config = {'width': 640, 'height': 480, 'fps': 30, 'use_picamera': True}
    camera = CameraModule(config)

    if camera.initialize():
        print("카메라 테스트 - 'q' 키로 종료")
        while True:
            ret, frame = camera.read_frame()
            if ret:
                cv2.imshow('Camera Test', frame)
                if cv2.waitKey(1) & 0xFF == ord('q'):
                    break
        camera.release()
        cv2.destroyAllWindows()
```

### 5.3 이미지 전처리 모듈 (preprocessor.py)

```python
# src/preprocessor.py
import cv2
import numpy as np

class ImagePreprocessor:
    def __init__(self, config):
        self.config = config

    def preprocess(self, image):
        """전체 전처리 파이프라인"""
        # 그레이스케일 변환
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)

        # 노이즈 제거
        blur = cv2.GaussianBlur(gray, self.config['blur_kernel'], 0)

        # 대비 향상 (CLAHE)
        clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
        enhanced = clahe.apply(blur)

        return enhanced

    def preprocess_for_detection(self, image):
        """번호판 검출용 전처리"""
        preprocessed = self.preprocess(image)

        # 엣지 검출
        edges = cv2.Canny(
            preprocessed,
            self.config['canny_threshold1'],
            self.config['canny_threshold2']
        )

        # 모폴로지 연산 (엣지 연결)
        kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (3, 3))
        edges = cv2.dilate(edges, kernel, iterations=1)

        return edges

    def preprocess_for_ocr(self, plate_image):
        """OCR용 전처리"""
        # 리사이즈 (높이 기준)
        height = 60
        aspect = plate_image.shape[1] / plate_image.shape[0]
        width = int(height * aspect)
        resized = cv2.resize(plate_image, (width, height))

        # 그레이스케일
        if len(resized.shape) == 3:
            gray = cv2.cvtColor(resized, cv2.COLOR_BGR2GRAY)
        else:
            gray = resized

        # 이진화 (Otsu)
        _, binary = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)

        # 배경이 검은색이면 반전
        if np.mean(binary) < 127:
            binary = cv2.bitwise_not(binary)

        # 노이즈 제거
        kernel = np.ones((2, 2), np.uint8)
        binary = cv2.morphologyEx(binary, cv2.MORPH_CLOSE, kernel)
        binary = cv2.morphologyEx(binary, cv2.MORPH_OPEN, kernel)

        # 패딩 추가 (OCR 정확도 향상)
        binary = cv2.copyMakeBorder(binary, 10, 10, 10, 10, cv2.BORDER_CONSTANT, value=255)

        return binary

    def deskew(self, image):
        """기울기 보정"""
        coords = np.column_stack(np.where(image > 0))
        if len(coords) < 5:
            return image

        angle = cv2.minAreaRect(coords)[-1]
        if angle < -45:
            angle = -(90 + angle)
        else:
            angle = -angle

        if abs(angle) < 0.5:
            return image

        (h, w) = image.shape[:2]
        center = (w // 2, h // 2)
        M = cv2.getRotationMatrix2D(center, angle, 1.0)
        rotated = cv2.warpAffine(image, M, (w, h),
                                  flags=cv2.INTER_CUBIC,
                                  borderMode=cv2.BORDER_REPLICATE)
        return rotated

# src/plate_detector.py
import cv2
import numpy as np
import os

class PlateDetector:
    def __init__(self, config, model_dir):
        self.config = config
        self.model_dir = model_dir
        self.method = config.get('method', 'contour')

        # Haar Cascade 로드 (method='haar' 인 경우)
        if self.method == 'haar':
            cascade_path = os.path.join(model_dir, 'haarcascade_russian_plate_number.xml')
            if os.path.exists(cascade_path):
                self.cascade = cv2.CascadeClassifier(cascade_path)
            else:
                print(f"[WARN] Cascade 파일 없음: {cascade_path}")
                self.method = 'contour'

    def detect(self, image, preprocessed):
        """번호판 검출"""
        if self.method == 'haar':
            return self._detect_haar(image)
        elif self.method == 'contour':
            return self._detect_contour(image, preprocessed)
        else:
            return self._detect_contour(image, preprocessed)

    def _detect_haar(self, image):
        """Haar Cascade 기반 검출"""
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY) if len(image.shape) == 3 else image
        plates = self.cascade.detectMultiScale(gray, scaleFactor=1.1, minNeighbors=5)

        results = []
        for (x, y, w, h) in plates:
            roi = image[y:y+h, x:x+w]
            results.append({
                'bbox': (x, y, w, h),
                'roi': roi,
                'confidence': 0.8
            })
        return results

    def _detect_contour(self, image, preprocessed):
        """컨투어 기반 검출"""
        # 컨투어 찾기
        contours, _ = cv2.findContours(preprocessed, cv2.RETR_TREE, cv2.CHAIN_APPROX_SIMPLE)

        candidates = []
        for contour in contours:
            area = cv2.contourArea(contour)

            # 면적 필터
            if area < self.config['min_area'] or area > self.config['max_area']:
                continue

            # 사각형 근사
            peri = cv2.arcLength(contour, True)
            approx = cv2.approxPolyDP(contour, 0.02 * peri, True)

            # 4~6개 꼭지점
            if len(approx) < 4 or len(approx) > 6:
                continue

            # 바운딩 박스
            x, y, w, h = cv2.boundingRect(contour)

            # 종횡비 필터 (번호판은 가로로 긴 형태)
            aspect_ratio = w / h
            if aspect_ratio < self.config['aspect_ratio_min'] or \
               aspect_ratio > self.config['aspect_ratio_max']:
                continue

            # 후보 추가
            roi = image[y:y+h, x:x+w]
            candidates.append({
                'bbox': (x, y, w, h),
                'roi': roi,
                'area': area,
                'aspect_ratio': aspect_ratio,
                'confidence': min(1.0, area / 10000)  # 면적 기반 신뢰도
            })

        # 면적 기준 상위 5개만 반환
        candidates.sort(key=lambda x: x['area'], reverse=True)
        return candidates[:5]

    def draw_detections(self, image, detections):
        """검출 결과 시각화"""
        result = image.copy()
        for det in detections:
            x, y, w, h = det['bbox']
            cv2.rectangle(result, (x, y), (x+w, y+h), (0, 255, 0), 2)

            # 신뢰도 표시
            conf = det.get('confidence', 0)
            cv2.putText(result, f"{conf:.2f}", (x, y-10),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)
        return result

# src/ocr_engine.py
import pytesseract
import cv2
import re

class OCREngine:
    def __init__(self, config):
        self.config = config
        self.lang = config.get('lang', 'kor+eng')
        self.psm = config.get('psm', 7)
        self.oem = config.get('oem', 3)
        self.whitelist = config.get('whitelist', '')

    def recognize(self, image):
        """문자 인식"""
        # Tesseract 설정
        custom_config = f'--oem {self.oem} --psm {self.psm}'
        if self.whitelist:
            custom_config += f' -c tessedit_char_whitelist={self.whitelist}'

        try:
            text = pytesseract.image_to_string(image, lang=self.lang, config=custom_config)
            text = self._clean_text(text)

            # 상세 정보 (신뢰도 포함)
            data = pytesseract.image_to_data(image, lang=self.lang,
                                             config=custom_config, output_type=pytesseract.Output.DICT)
            confidence = self._calculate_confidence(data)

            return {
                'text': text,
                'confidence': confidence,
                'raw_data': data
            }
        except Exception as e:
            print(f"[ERROR] OCR 실패: {e}")
            return {'text': '', 'confidence': 0, 'raw_data': None}

    def _clean_text(self, text):
        """텍스트 정리"""
        # 공백 및 특수문자 제거
        text = re.sub(r'\s+', '', text)
        text = re.sub(r'[^\w가-힣]', '', text)
        return text.strip()

    def _calculate_confidence(self, data):
        """평균 신뢰도 계산"""
        confidences = [int(c) for c in data['conf'] if int(c) > 0]
        if confidences:
            return sum(confidences) / len(confidences)
        return 0


class PlateValidator:
    def __init__(self, patterns):
        self.patterns = [re.compile(p) for p in patterns]

    def validate(self, text):
        """번호판 형식 검증"""
        for pattern in self.patterns:
            if pattern.match(text):
                return True, pattern.pattern
        return False, None

    def extract_plate_number(self, text):
        """번호판 번호 추출 시도"""
        # 숫자와 한글만 추출
        cleaned = re.sub(r'[^0-9가-힣]', '', text)

        # 일반적인 번호판 패턴 매칭
        patterns = [
            r'(\d{2,3}[가-힣]\d{4})',           # 12가1234, 123가1234
            r'([가-힣]{2}\d{2}[가-힣]\d{4})',    # 서울12가1234
        ]

        for pattern in patterns:
            match = re.search(pattern, cleaned)
            if match:
                return match.group(1)

        return cleaned if cleaned else None

# main.py
import cv2
import time
import os
import sys
from datetime import datetime

# 모듈 경로 추가
sys.path.insert(0, os.path.join(os.path.dirname(__file__), 'src'))

from src.camera import CameraModule
from src.preprocessor import ImagePreprocessor
from src.plate_detector import PlateDetector
from src.ocr_engine import OCREngine, PlateValidator
import config

class ANPRSystem:
    def __init__(self):
        # 모듈 초기화
        self.camera = CameraModule(config.CAMERA_CONFIG)
        self.preprocessor = ImagePreprocessor(config.PREPROCESS_CONFIG)
        self.detector = PlateDetector(config.PLATE_DETECT_CONFIG, config.MODEL_DIR)
        self.ocr = OCREngine(config.OCR_CONFIG)
        self.validator = PlateValidator(config.PLATE_PATTERNS)

        # 로그 디렉토리 생성
        os.makedirs(config.LOG_DIR, exist_ok=True)

        # 최근 인식 결과 (중복 방지)
        self.recent_plates = {}
        self.duplicate_timeout = 5  # 초

    def initialize(self):
        """시스템 초기화"""
        print("=" * 50)
        print("ANPR System 초기화 중...")
        print("=" * 50)

        if not self.camera.initialize():
            return False

        print("[INFO] 시스템 준비 완료")
        return True

    def process_frame(self, frame):
        """단일 프레임 처리"""
        results = []

        # 전처리
        preprocessed = self.preprocessor.preprocess_for_detection(frame)

        # 번호판 검출
        detections = self.detector.detect(frame, preprocessed)

        for det in detections:
            roi = det['roi']

            # OCR 전처리
            ocr_ready = self.preprocessor.preprocess_for_ocr(roi)

            # 문자 인식
            ocr_result = self.ocr.recognize(ocr_ready)
            text = ocr_result['text']
            confidence = ocr_result['confidence']

            if not text:
                continue

            # 번호판 형식 검증
            is_valid, _ = self.validator.validate(text)

            # 중복 체크
            if self._is_duplicate(text):
                continue

            result = {
                'text': text,
                'confidence': confidence,
                'is_valid': is_valid,
                'bbox': det['bbox'],
                'timestamp': datetime.now().isoformat()
            }
            results.append(result)

            # 유효한 번호판이면 로그
            if is_valid and confidence > 50:
                self._log_detection(result, frame, roi)

        return results, detections

    def _is_duplicate(self, text):
        """중복 체크"""
        current_time = time.time()

        # 오래된 항목 제거
        expired = [k for k, v in self.recent_plates.items()
                   if current_time - v > self.duplicate_timeout]
        for k in expired:
            del self.recent_plates[k]

        # 중복 확인
        if text in self.recent_plates:
            return True

        self.recent_plates[text] = current_time
        return False

    def _log_detection(self, result, frame, roi):
        """검출 로그 저장"""
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")

        # 콘솔 출력
        print(f"\n{'='*40}")
        print(f"[번호판 인식] {result['text']}")
        print(f"신뢰도: {result['confidence']:.1f}%")
        print(f"시간: {result['timestamp']}")
        print(f"{'='*40}\n")

        # 이미지 저장
        if config.LOG_CONFIG.get('save_images', False):
            frame_path = os.path.join(config.LOG_DIR, f"{timestamp}_frame.jpg")
            roi_path = os.path.join(config.LOG_DIR, f"{timestamp}_plate.jpg")
            cv2.imwrite(frame_path, frame)
            cv2.imwrite(roi_path, roi)

    def run_realtime(self):
        """실시간 인식 모드"""
        print("\n실시간 인식 시작 - 'q'키로 종료")

        fps_counter = 0
        fps_start_time = time.time()
        fps = 0

        while True:
            ret, frame = self.camera.read_frame()
            if not ret:
                print("[ERROR] 프레임 읽기 실패")
                break

            # 프레임 처리
            results, detections = self.process_frame(frame)

            # 검출 결과 시각화
            display = self.detector.draw_detections(frame, detections)

            # 인식 결과 표시
            for result in results:
                x, y, w, h = result['bbox']
                text = f"{result['text']} ({result['confidence']:.0f}%)"
                color = (0, 255, 0) if result['is_valid'] else (0, 165, 255)
                cv2.putText(display, text, (x, y + h + 20),
                           cv2.FONT_HERSHEY_SIMPLEX, 0.6, color, 2)

            # FPS 계산
            fps_counter += 1
            if time.time() - fps_start_time >= 1:
                fps = fps_counter
                fps_counter = 0
                fps_start_time = time.time()

            cv2.putText(display, f"FPS: {fps}", (10, 30),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 255), 2)

            cv2.imshow('ANPR System', display)

            if cv2.waitKey(1) & 0xFF == ord('q'):
                break

        self.cleanup()

    def run_single_image(self, image_path):
        """단일 이미지 인식 모드"""
        print(f"\n이미지 인식: {image_path}")

        frame = cv2.imread(image_path)
        if frame is None:
            print(f"[ERROR] 이미지 로드 실패: {image_path}")
            return

        results, detections = self.process_frame(frame)

        # 결과 출력
        print(f"\n검출된 후보: {len(detections)}개")
        print(f"인식된 번호판: {len(results)}개")

        for i, result in enumerate(results):
            print(f"\n[결과 {i+1}]")
            print(f"  텍스트: {result['text']}")
            print(f"  신뢰도: {result['confidence']:.1f}%")
            print(f"  유효 번호판: {result['is_valid']}")

        # 시각화
        display = self.detector.draw_detections(frame, detections)
        cv2.imshow('Result', display)
        cv2.waitKey(0)
        cv2.destroyAllWindows()

    def cleanup(self):
        """정리"""
        self.camera.release()
        cv2.destroyAllWindows()
        print("\n시스템 종료")


def main():
    import argparse

    parser = argparse.ArgumentParser(description='ANPR System')
    parser.add_argument('--mode', choices=['realtime', 'image'], default='realtime',
                       help='실행 모드 (realtime: 실시간, image: 단일 이미지)')

    except KeyboardInterrupt:

    finally:

if __name__ == "__main__":
    main()

    # 배경 이미지 (도로 시뮬레이션)

def test_detection():
    """검출 테스트"""
    preprocess_config = {
        'blur_kernel': (5, 5),
        'canny_threshold1': 100,
        'canny_threshold2': 200,
    }

    detect_config = {
        'method': 'contour',
        'min_area': 500,
        'max_area': 50000,
        'aspect_ratio_min': 2.0,
        'aspect_ratio_max': 6.0,
    }

    preprocessor = ImagePreprocessor(preprocess_config)
    detector = PlateDetector(detect_config, '../models')

    # 합성 이미지 생성
    img, true_bbox = create_synthetic_plate_image()

    # 전처리
    preprocessed = preprocessor.preprocess_for_detection(img)

    # 검출
    detections = detector.detect(img, preprocessed)

    print("\n[번호판 검출 테스트]")
    print(f"  실제 위치: {true_bbox}")
    print(f"  검출된 후보: {len(detections)}개")

    # IoU 계산 (정확도 측정)
    for i, det in enumerate(detections):
        iou = calculate_iou(true_bbox, det['bbox'])
        print(f"  후보 {i+1}: bbox={det['bbox']}, IoU={iou:.2f}")

    # 시각화
    result = detector.draw_detections(img, detections)
    cv2.imshow('Detection Test', result)
    cv2.waitKey(0)
    cv2.destroyAllWindows()

def calculate_iou(box1, box2):
    """IoU (Intersection over Union) 계산"""
    x1, y1, w1, h1 = box1
    x2, y2, w2, h2 = box2

    # 교집합 영역
    xi1 = max(x1, x2)
    yi1 = max(y1, y2)
    xi2 = min(x1 + w1, x2 + w2)
    yi2 = min(y1 + h1, y2 + h2)

    if xi2 <= xi1 or yi2 <= yi1:
        return 0.0

    inter_area = (xi2 - xi1) * (yi2 - yi1)

    # 합집합 영역
    box1_area = w1 * h1
    box2_area = w2 * h2
    union_area = box1_area + box2_area - inter_area

    return inter_area / union_area if union_area > 0 else 0.0

if __name__ == "__main__":
    test_detection()

import threading
from queue import Queue

class ThreadedCamera:
    def __init__(self, camera):
        self.camera = camera
        self.frame_queue = Queue(maxsize=2)
        self.stopped = False

    def start(self):
        threading.Thread(target=self._update, daemon=True).start()
        return self

    def _update(self):
        while not self.stopped:
            ret, frame = self.camera.read_frame()
            if ret:
                if self.frame_queue.full():
                    self.frame_queue.get()
                self.frame_queue.put(frame)

    def read(self):
        return self.frame_queue.get()

    def stop(self):
        self.stopped = True

        })
        self.client.publish('anpr/plates', payload)
```
