## 5. 추천 알고리즘

### 5.1 총점 계산

```javascript
// 서울대 정시 반영비율 적용
function calculateTotalScore(korean, math, explore) {
    const koreanWeighted = korean * 0.333;      // 국어 33.3%
    const mathWeighted = math * 1.2 * 0.333;    // 수학 40% (가중치 1.2)
    const exploreWeighted = explore * 0.8 * 0.333; // 탐구 26.7% (가중치 0.8)

    return koreanWeighted + mathWeighted + exploreWeighted;
}
```

### 5.2 매칭 레벨 분류

| 레벨 | 점수 차이 | 합격 가능성 | 설명 |
|------|---------|-----------|------|
| **안정 (Safe)** | +3점 이상 | 70~95% | 커트라인 대비 여유 |
| **적정 (Match)** | -2 ~ +3점 | 40~70% | 실력에 적합 |
| **상향 (Challenge)** | -5 ~ -2점 | 20~40% | 도전 가능 |
| **어려움 (Unlikely)** | -5점 미만 | 20% 미만 | 추천 제외 |

### 5.3 매칭 점수 계산

```javascript
function analyzeDepartment(userScore, cutline) {
    // 각 영역별 차이 계산
    const koreanDiff = userScore.korean - cutline.korean;
    const mathDiff = userScore.math - cutline.math;
    const exploreDiff = userScore.explore - cutline.explore;

    // 가중 평균 점수 차이
    const score = (koreanDiff * 0.333) + (mathDiff * 0.4) + (exploreDiff * 0.267);

    // 영어/내신 조건 확인
    const englishOK = userScore.english <= cutline.english;
    const naesinOK = userScore.naesin <= cutline.naesin;

    // 매칭 레벨 결정
    if (score >= 3 && englishOK && naesinOK) return 'safe';
    if (score >= -2 && score < 3) return 'match';
    if (score >= -5) return 'challenge';
    return 'unlikely';
}
```

### 5.4 합격 확률 계산

```javascript
function calculateMatchPercent(score, englishOK, naesinOK) {
    let percent;

    if (score >= 3 && englishOK && naesinOK) {
        percent = Math.min(95, 70 + score * 3);  // 안정: 70~95%
    } else if (score >= -2 && score < 3) {
        percent = Math.min(70, 50 + score * 5);  // 적정: 40~70%
    } else if (score >= -5) {
        percent = Math.max(20, 40 + score * 5);  // 상향: 20~40%
    } else {
        percent = Math.max(5, 20 + score * 3);   // 어려움: 5~20%
    }

    return Math.round(percent);
}
```
