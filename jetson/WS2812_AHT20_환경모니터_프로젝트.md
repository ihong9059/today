# Jetson Nano 환경 모니터 프로젝트

**WS2812 LED + AHT20 온습도 센서를 이용한 스마트 환경 모니터**

---

## 1. 프로젝트 개요

### 1.1 목표

Jetson Nano의 GPIO를 활용하여 온습도 센서(AHT20)로 환경 데이터를 수집하고,
네오픽셀 LED(WS2812)로 시각적으로 표현하는 "스마트 환경 모니터" 제작

### 1.2 프로젝트 컨셉

```
┌─────────────────────────────────────────────────────────────┐
│                    스마트 환경 모니터                         │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│   [AHT20 센서]                      [WS2812 LED Strip]      │
│       │                                   │                 │
│   온도: 25.3°C  ──────►  LED 1-4: 온도 표시 (파랑↔빨강)      │
│   습도: 65.2%   ──────►  LED 5-8: 습도 표시 (노랑↔청록)      │
│                                                             │
│   ┌─────────────────────────────────────────┐              │
│   │  [파][파][초][초][노][노][청][청]        │              │
│   │   온도 영역      │     습도 영역         │              │
│   └─────────────────────────────────────────┘              │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

### 1.3 학습 목표

| 항목 | 내용 |
|:-----|:-----|
| **I2C 통신** | AHT20 센서와 I2C 프로토콜 통신 |
| **GPIO 제어** | WS2812 LED 제어 (PWM/SPI) |
| **데이터 처리** | 센서값 → 시각화 변환 로직 |
| **Python 프로그래밍** | Jetson.GPIO, 라이브러리 활용 |

### 2.3 배선도

#### AHT20 연결 (I2C)

| AHT20 핀 | Jetson Nano 핀 | 설명 |
|:---------|:---------------|:-----|
| VCC | Pin 4 (5V) | 전원 |
| GND | Pin 6 (GND) | 접지 |
| SDA | Pin 3 (SDA) | I2C 데이터 |
| SCL | Pin 5 (SCL) | I2C 클럭 |

#### WS2812 연결 (PWM)

| WS2812 핀 | Jetson Nano 핀 | 설명 |
|:----------|:---------------|:-----|
| VCC | 외부 5V 전원 | LED 전원 (별도 공급 권장) |
| GND | Pin 14 (GND) + 외부 GND | 공통 접지 |
| DIN | Pin 12 (GPIO18) | 데이터 신호 |

### 4.2 메인 앱 코드 (env_monitor.py)

```python
#!/usr/bin/env python3
"""
Jetson Nano 환경 모니터
- AHT20: 온습도 센서 (I2C)
- WS2812: LED 시각화 (PWM)
"""

import time
import board
import adafruit_ahtx0
import neopixel

# ===== 설정 =====
LED_COUNT = 8           # WS2812 LED 개수
LED_PIN = board.D18     # GPIO18 (Pin 12)
LED_BRIGHTNESS = 0.3    # 밝기 (0.0 ~ 1.0)

# 온도 범위 (°C)
TEMP_MIN = 15.0
TEMP_MAX = 35.0

# 습도 범위 (%)
HUMID_MIN = 30.0
HUMID_MAX = 80.0

# ===== 초기화 =====
def init_sensor():
    """AHT20 센서 초기화"""
    i2c = board.I2C()
    sensor = adafruit_ahtx0.AHTx0(i2c)
    return sensor

def init_led():
    """WS2812 LED 초기화"""
    pixels = neopixel.NeoPixel(
        LED_PIN,
        LED_COUNT,
        brightness=LED_BRIGHTNESS,
        auto_write=False
    )
    return pixels

# ===== 색상 변환 =====
def temp_to_color(temp):
    """
    온도를 색상으로 변환
    - 낮은 온도: 파랑 (0, 0, 255)
    - 높은 온도: 빨강 (255, 0, 0)
    """
    # 온도를 0~1 범위로 정규화
    ratio = (temp - TEMP_MIN) / (TEMP_MAX - TEMP_MIN)
    ratio = max(0, min(1, ratio))  # 0~1 클램핑

    # 파랑 → 초록 → 빨강 그라데이션
    if ratio < 0.5:
        # 파랑 → 초록
        r = 0
        g = int(255 * (ratio * 2))
        b = int(255 * (1 - ratio * 2))
    else:
        # 초록 → 빨강
        r = int(255 * ((ratio - 0.5) * 2))
        g = int(255 * (1 - (ratio - 0.5) * 2))
        b = 0

    return (r, g, b)

def humid_to_color(humid):
    """
    습도를 색상으로 변환
    - 낮은 습도: 노랑 (255, 200, 0)
    - 높은 습도: 청록 (0, 200, 255)
    """
    ratio = (humid - HUMID_MIN) / (HUMID_MAX - HUMID_MIN)
    ratio = max(0, min(1, ratio))

    r = int(255 * (1 - ratio))
    g = 200
    b = int(255 * ratio)

    return (r, g, b)

# ===== LED 업데이트 =====
def update_leds(pixels, temp, humid):
    """
    LED 색상 업데이트
    - LED 0-3: 온도 표시
    - LED 4-7: 습도 표시
    """
    temp_color = temp_to_color(temp)
    humid_color = humid_to_color(humid)

    # 온도 LED (0-3)
    for i in range(4):
        pixels[i] = temp_color

    # 습도 LED (4-7)
    for i in range(4, 8):
        pixels[i] = humid_color

    pixels.show()

# ===== 메인 =====
def main():
    print("=" * 50)
    print("  Jetson Nano 환경 모니터")
    print("  AHT20 + WS2812")
    print("=" * 50)

    # 초기화
    print("\n[초기화]")
    try:
        sensor = init_sensor()
        print("  ✓ AHT20 센서 연결됨")
    except Exception as e:
        print(f"  ✗ AHT20 연결 실패: {e}")
        return

    try:
        pixels = init_led()
        print("  ✓ WS2812 LED 연결됨")
    except Exception as e:
        print(f"  ✗ WS2812 연결 실패: {e}")
        return

    print("\n[모니터링 시작] (Ctrl+C로 종료)")
    print("-" * 50)

    try:
        while True:
            # 센서 읽기
            temp = sensor.temperature
            humid = sensor.relative_humidity

            # LED 업데이트
            update_leds(pixels, temp, humid)

            # 콘솔 출력
            print(f"\r  온도: {temp:5.1f}°C | 습도: {humid:5.1f}%", end="")

            time.sleep(1)

    except KeyboardInterrupt:
        print("\n\n[종료]")
        # LED 끄기
        pixels.fill((0, 0, 0))
        pixels.show()
        print("  모니터링 종료됨")

if __name__ == "__main__":
    main()
```

### 4.3 AHT20 테스트 코드 (test_aht20.py)

```python
#!/usr/bin/env python3
"""AHT20 센서 테스트"""

import time
import board
import adafruit_ahtx0

def main():
    print("AHT20 센서 테스트")
    print("=" * 40)

    # I2C 초기화
    i2c = board.I2C()
    sensor = adafruit_ahtx0.AHTx0(i2c)

    print("센서 연결 성공!")
    print("5초간 측정합니다...\n")

    for i in range(5):
        temp = sensor.temperature
        humid = sensor.relative_humidity
        print(f"[{i+1}] 온도: {temp:.2f}°C, 습도: {humid:.2f}%")
        time.sleep(1)

    print("\n테스트 완료!")

if __name__ == "__main__":
    main()
```

### 4.4 WS2812 테스트 코드 (test_ws2812.py)

```python
#!/usr/bin/env python3
"""WS2812 LED 테스트"""

import time
import board
import neopixel

LED_COUNT = 8
LED_PIN = board.D18

def main():
    print("WS2812 LED 테스트")
    print("=" * 40)

    pixels = neopixel.NeoPixel(LED_PIN, LED_COUNT, brightness=0.3, auto_write=False)

    # 테스트 1: 전체 빨강
    print("테스트 1: 빨강")
    pixels.fill((255, 0, 0))
    pixels.show()
    time.sleep(1)

    # 테스트 2: 전체 초록
    print("테스트 2: 초록")
    pixels.fill((0, 255, 0))
    pixels.show()
    time.sleep(1)

    # 테스트 3: 전체 파랑
    print("테스트 3: 파랑")
    pixels.fill((0, 0, 255))
    pixels.show()
    time.sleep(1)

    # 테스트 4: 레인보우
    print("테스트 4: 레인보우")
    colors = [
        (255, 0, 0),    # 빨강
        (255, 127, 0),  # 주황
        (255, 255, 0),  # 노랑
        (0, 255, 0),    # 초록
        (0, 0, 255),    # 파랑
        (75, 0, 130),   # 남색
        (148, 0, 211),  # 보라
        (255, 255, 255) # 흰색
    ]
    for i, color in enumerate(colors):
        pixels[i] = color
    pixels.show()
    time.sleep(2)

    # 종료: LED 끄기
    print("테스트 완료!")
    pixels.fill((0, 0, 0))
    pixels.show()

if __name__ == "__main__":
    main()
```
