## 3. 실습 상세 - 핵심 실습 7가지

### 실습 1: "공장 신호등" (Day 2)

```python
# 파일: factory_signal.py
# 센서값에 따라 LED 신호등 제어

import RPi.GPIO as GPIO
import Adafruit_DHT
import time

# 핀 설정
LED_GREEN = 17   # 정상
LED_YELLOW = 27  # 주의
LED_RED = 22     # 위험
BUZZER = 23
SENSOR = Adafruit_DHT.DHT22
SENSOR_PIN = 4

GPIO.setmode(GPIO.BCM)
for pin in [LED_GREEN, LED_YELLOW, LED_RED, BUZZER]:
    GPIO.setup(pin, GPIO.OUT)

try:
    while True:
        humidity, temperature = Adafruit_DHT.read_retry(SENSOR, SENSOR_PIN)
        print(f"온도: {temperature:.1f}C, 습도: {humidity:.1f}%")

        # 신호등 로직
        GPIO.output(LED_GREEN, temperature < 28)
        GPIO.output(LED_YELLOW, 28 <= temperature < 35)
        GPIO.output(LED_RED, temperature >= 35)
        GPIO.output(BUZZER, temperature >= 35)

        time.sleep(2)
except KeyboardInterrupt:
    GPIO.cleanup()
```

**학습 포인트**: GPIO 출력, 센서 읽기, 조건 분기, 무한루프

### 실습 2: "진동 이상탐지 AI" (Day 4)

```python
# 파일: vibration_anomaly.py
# 진동 센서 데이터에서 이상 패턴 자동 탐지

import numpy as np
from sklearn.ensemble import IsolationForest
import json, time

# 진동 데이터 수집 (실제로는 ADXL345에서 읽음)
def read_vibration():
    # Shield의 ADXL345 가속도 센서에서 x,y,z 읽기
    import board, busio, adafruit_adxl345
    i2c = busio.I2C(board.SCL, board.SDA)
    sensor = adafruit_adxl345.ADXL345(i2c)
    x, y, z = sensor.acceleration
    magnitude = np.sqrt(x**2 + y**2 + z**2)
    return magnitude

# 1단계: 정상 데이터 100개 수집 (30초)
print("정상 운전 데이터 수집 중... (설비를 정상 가동하세요)")
normal_data = []
for i in range(100):
    normal_data.append([read_vibration()])
    time.sleep(0.3)

# 2단계: AI 모델 학습
model = IsolationForest(contamination=0.05, random_state=42)
model.fit(normal_data)
print("AI 학습 완료! 이상 감시를 시작합니다.")

# 3단계: 실시간 감시
while True:
    value = read_vibration()
    result = model.predict([[value]])  # 1=정상, -1=이상
    status = "정상" if result[0] == 1 else "이상 감지!"
    print(f"진동: {value:.2f} → {status}")
    time.sleep(0.5)
```

**학습 포인트**: 데이터 수집 → 모델 학습 → 실시간 추론의 AI 파이프라인 이해

### 실습 3: "카메라 불량 검사" (Day 4)

```python
# 파일: vision_inspection.py
# 카메라로 부품 색상 검사 (빨강=불량, 초록=양품)

import cv2
import numpy as np

cap = cv2.VideoCapture(0)

def check_quality(frame):
    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)

    # 빨간색 영역 (불량)
    red_mask = cv2.inRange(hsv, (0,100,100), (10,255,255))
    red_area = cv2.countNonZero(red_mask)

    # 초록색 영역 (양품)
    green_mask = cv2.inRange(hsv, (35,100,100), (85,255,255))
    green_area = cv2.countNonZero(green_mask)

    if red_area > 5000:
        return "FAIL", (0, 0, 255)
    elif green_area > 5000:
        return "PASS", (0, 255, 0)
    else:
        return "UNKNOWN", (128, 128, 128)

while True:
    ret, frame = cap.read()
    result, color = check_quality(frame)
    cv2.putText(frame, result, (50, 80),
                cv2.FONT_HERSHEY_SIMPLEX, 2, color, 3)
    cv2.imshow("Quality Inspection", frame)
    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

**학습 포인트**: OpenCV 이미지 처리, HSV 색 공간, 실시간 비전 판정
