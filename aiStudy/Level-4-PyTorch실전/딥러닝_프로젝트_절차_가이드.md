## 전체 워크플로우 개요

```
┌─────────────────────────────────────────────────────────────────┐
│                    딥러닝 프로젝트 파이프라인                      │
├─────────────────────────────────────────────────────────────────┤
│  1. 데이터 수집    →  이미지, 텍스트, 센서 데이터 등 원본 수집      │
│  2. 데이터 정리    →  라벨링, 폴더 구조화, 품질 검수               │
│  3. Dataset 클래스 →  PyTorch가 이해하는 형식으로 변환            │
│  4. DataLoader    →  배치 단위로 데이터 공급                     │
│  5. 모델 정의     →  신경망 구조 설계 (nn.Module)                │
│  6. 학습 설정     →  손실함수, 옵티마이저, 하이퍼파라미터          │
│  7. 학습 루프     →  순전파 → 손실계산 → 역전파 → 가중치 업데이트   │
│  8. 평가/검증     →  테스트셋으로 성능 측정                       │
│  9. 모델 저장     →  .pth 파일로 가중치 저장                     │
│  10. 배포/추론    →  실제 서비스에 적용                          │
└─────────────────────────────────────────────────────────────────┘
```

## Step 3: PyTorch Dataset 클래스 만들기

### 방법 1: ImageFolder 사용 (폴더 구조가 있을 때)

```python
from torchvision import datasets, transforms
from torch.utils.data import DataLoader

# 데이터 전처리 정의
transform = transforms.Compose([
    transforms.Resize((224, 224)),      # 이미지 크기 통일
    transforms.ToTensor(),              # 텐서로 변환

# ImageFolder로 데이터셋 생성 (폴더명이 자동으로 라벨이 됨)
```

### 방법 2: 커스텀 Dataset 클래스 (CSV 라벨링)

```python
import torch
from torch.utils.data import Dataset
from PIL import Image
import pandas as pd
import os

class CustomImageDataset(Dataset):
    """
    커스텀 이미지 데이터셋 클래스

    필수 메서드:
    - __init__: 초기화 (파일 경로, 라벨 로드)
    - __len__: 데이터셋 크기 반환
    - __getitem__: 인덱스로 샘플 반환
    """

    def __init__(self, csv_file, img_dir, transform=None):
        """
        Args:
            csv_file: 라벨 정보가 담긴 CSV 파일 경로
            img_dir: 이미지 폴더 경로
            transform: 이미지 변환 함수
        """
        self.labels_df = pd.read_csv(csv_file)
        self.img_dir = img_dir
        self.transform = transform

        # 라벨을 숫자로 변환
        self.label_to_idx = {
            label: idx for idx, label in
            enumerate(self.labels_df['label'].unique())
        }

    def __len__(self):
        """데이터셋의 총 샘플 수"""
        return len(self.labels_df)

    def __getitem__(self, idx):
        """
        인덱스에 해당하는 샘플 반환

        Returns:
            image: 변환된 이미지 텐서
            label: 정수형 라벨
        """
        # 이미지 경로
        img_name = self.labels_df.iloc[idx]['filename']
        img_path = os.path.join(self.img_dir, img_name)

        # 이미지 로드
        image = Image.open(img_path).convert('RGB')

        # 라벨
        label_str = self.labels_df.iloc[idx]['label']
        label = self.label_to_idx[label_str]

        # 변환 적용
        if self.transform:
            image = self.transform(image)

        return image, label

# 사용 예시
dataset = CustomImageDataset(
    csv_file='data/labels.csv',
    img_dir='data/images',
    transform=transform
)
```

### 방법 3: 다양한 데이터 타입

```python
# 텍스트 데이터
class TextDataset(Dataset):
    def __init__(self, texts, labels, tokenizer, max_length=128):
        self.texts = texts
        self.labels = labels
        self.tokenizer = tokenizer
        self.max_length = max_length

    def __len__(self):
        return len(self.texts)

    def __getitem__(self, idx):
        text = self.texts[idx]
        label = self.labels[idx]

        # 토큰화
        encoded = self.tokenizer(
            text,
            padding='max_length',
            truncation=True,
            max_length=self.max_length,
            return_tensors='pt'
        )

        return {
            'input_ids': encoded['input_ids'].squeeze(),
            'attention_mask': encoded['attention_mask'].squeeze(),
            'label': torch.tensor(label)
        }

# 수치 데이터 (센서, 주가 등)
class NumericDataset(Dataset):
    def __init__(self, features, labels):
        self.features = torch.FloatTensor(features)
        self.labels = torch.LongTensor(labels)

    def __len__(self):
        return len(self.labels)

    def __getitem__(self, idx):
        return self.features[idx], self.labels[idx]
```

```python
from torch.utils.data import DataLoader, random_split
```

## Step 5: 모델 정의

### 간단한 CNN 모델

```python
import torch.nn as nn

class SimpleCNN(nn.Module):
    def __init__(self, num_classes):
        super().__init__()

        # 특징 추출 레이어
        self.features = nn.Sequential(
            nn.Conv2d(3, 32, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2),

            nn.Conv2d(32, 64, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2),

            nn.Conv2d(64, 128, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2),
        )

        # 분류 레이어
        self.classifier = nn.Sequential(
            nn.Flatten(),
            nn.Linear(128 * 28 * 28, 512),  # 224 → 112 → 56 → 28
            nn.ReLU(),
            nn.Dropout(0.5),
            nn.Linear(512, num_classes)
        )

    def forward(self, x):
        x = self.features(x)
        x = self.classifier(x)
        return x

model = SimpleCNN(num_classes=3)  # cat, dog, bird
```

## Step 7: 학습 루프

```python
def train_one_epoch(model, dataloader, criterion, optimizer, device):
    model.train()
    running_loss = 0.0
    correct = 0
    total = 0

    for images, labels in dataloader:
        images, labels = images.to(device), labels.to(device)

        # 순전파
        optimizer.zero_grad()
        outputs = model(images)
        loss = criterion(outputs, labels)

        # 역전파
        loss.backward()
        optimizer.step()

        # 통계
        running_loss += loss.item()
        _, predicted = torch.max(outputs, 1)
        total += labels.size(0)
        correct += (predicted == labels).sum().item()

    epoch_loss = running_loss / len(dataloader)
    epoch_acc = 100 * correct / total
    return epoch_loss, epoch_acc

def validate(model, dataloader, criterion, device):
    model.eval()
    running_loss = 0.0
    correct = 0
    total = 0

    with torch.no_grad():
        for images, labels in dataloader:
            images, labels = images.to(device), labels.to(device)
            outputs = model(images)
            loss = criterion(outputs, labels)

            running_loss += loss.item()
            _, predicted = torch.max(outputs, 1)
            total += labels.size(0)
            correct += (predicted == labels).sum().item()

    epoch_loss = running_loss / len(dataloader)
    epoch_acc = 100 * correct / total
    return epoch_loss, epoch_acc

# 학습 실행
print("학습 시작...")
print("-" * 60)

for epoch in range(epochs):
    train_loss, train_acc = train_one_epoch(
        model, train_loader, criterion, optimizer, device
    )
    val_loss, val_acc = validate(
        model, val_loader, criterion, device
    )

    # 학습률 스케줄러 업데이트
    scheduler.step()

    print(f"Epoch {epoch+1:2d}/{epochs} | "
          f"Train Loss: {train_loss:.4f}, Acc: {train_acc:.2f}% | "
          f"Val Loss: {val_loss:.4f}, Acc: {val_acc:.2f}%")

    # 최고 성능 모델 저장
    if val_acc > best_accuracy:
        best_accuracy = val_acc
        torch.save(model.state_dict(), "best_model.pth")
        print(f"  → 새로운 최고 성능! 모델 저장됨")

print("-" * 60)
print(f"학습 완료! 최고 검증 정확도: {best_accuracy:.2f}%")
```

## Step 10: 추론 (실제 사용)

```python
# 모델 로드
model = SimpleCNN(num_classes=3)
model.load_state_dict(torch.load("best_model.pth"))
model.eval()

# 새 이미지 예측
from PIL import Image

def predict(image_path, model, transform, class_names):
    # 이미지 로드 및 전처리
    image = Image.open(image_path).convert('RGB')
    image_tensor = transform(image).unsqueeze(0)  # 배치 차원 추가

    # 예측
    with torch.no_grad():
        output = model(image_tensor)
        probabilities = torch.softmax(output, dim=1)
        predicted_idx = torch.argmax(output, dim=1).item()
        confidence = probabilities[0][predicted_idx].item()

    predicted_class = class_names[predicted_idx]
    return predicted_class, confidence

# 사용 예시
class_names = ['bird', 'cat', 'dog']
result, conf = predict("test_image.jpg", model, transform, class_names)
print(f"예측: {result} (확신도: {conf*100:.1f}%)")
```

## MNIST vs 커스텀 프로젝트 비교

| 항목 | MNIST | 커스텀 프로젝트 |
|------|-------|----------------|
| 데이터 수집 | 자동 다운로드 | 직접 수집/구매 |
| 라벨링 | 이미 완료됨 | 직접 라벨링 필요 |
| 전처리 | 표준화됨 | 데이터에 맞게 설계 |
| Dataset | datasets.MNIST | 커스텀 클래스 작성 |
| 모델 | 간단한 MLP | 문제에 맞는 구조 |
| 하이퍼파라미터 | 참고 자료 풍부 | 직접 실험 필요 |
| 평가 | 벤치마크 존재 | 자체 기준 설정 |
