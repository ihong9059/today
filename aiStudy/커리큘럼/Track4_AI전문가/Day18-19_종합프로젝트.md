```python
# src/detector.py
from ultralytics import YOLO

class PlateDetector:
    def __init__(self, model_path, conf_threshold=0.5):
        self.model = YOLO(model_path)
        self.conf_threshold = conf_threshold

    def detect(self, image):
        results = self.model(image, verbose=False, conf=self.conf_threshold)
        detections = []
        for box in results[0].boxes:
            x1, y1, x2, y2 = map(int, box.xyxy[0].tolist())
            detections.append({
                'bbox': (x1, y1, x2, y2),
                'confidence': float(box.conf),
            })
        return detections
```

```python
# src/preprocessor.py
import cv2
import numpy as np

class PlatePreprocessor:
    def __init__(self, method="standard"):
        self.method = method

    def process(self, plate_image):
        if len(plate_image.shape) == 3:
            gray = cv2.cvtColor(plate_image, cv2.COLOR_BGR2GRAY)
        else:
            gray = plate_image.copy()

        if self.method == "standard":
            denoised = cv2.bilateralFilter(gray, 11, 17, 17)
            _, binary = cv2.threshold(denoised, 0, 255,
                                       cv2.THRESH_BINARY + cv2.THRESH_OTSU)
            return binary
        elif self.method == "adaptive":
            clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
            enhanced = clahe.apply(gray)
            binary = cv2.adaptiveThreshold(enhanced, 255,
                                            cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
                                            cv2.THRESH_BINARY, 11, 2)
            return binary
        return gray
```

```python
# src/recognizer.py
import easyocr

class PlateRecognizer:
    def __init__(self, languages=['ko', 'en'], gpu=True):
        self.reader = easyocr.Reader(languages, gpu=gpu)

    def recognize(self, processed_image):
        results = self.reader.readtext(processed_image, detail=1)
        if not results:
            return "", 0.0
        best = max(results, key=lambda x: x[2])
        return best[1], float(best[2])
```

```python
# src/postprocessor.py
import re

class PlatePostprocessor:
    PATTERNS = [
        (r'^\d{3}[가-힣]\d{4}$', '신형'),
        (r'^\d{2}[가-힣]\d{4}$', '구형'),
        (r'^[가-힣]{2}\d{2}[가-힣]\d{4}$', '영업용'),
    ]
    CORRECTIONS = {
        'O': '0', 'o': '0', 'I': '1', 'l': '1',
        'Z': '2', 'S': '5', 'B': '8', 'G': '6',
    }

    def process(self, raw_text):
        cleaned = re.sub(r'[^0-9가-힣a-zA-Z]', '', raw_text)
        corrected = []
        for ch in cleaned:
            corrected.append(self.CORRECTIONS.get(ch, ch))
        corrected = ''.join(corrected)

        for pattern, plate_type in self.PATTERNS:
            if re.match(pattern, corrected):
                return corrected, True, plate_type
        return corrected, False, "미인식"
```

```python
# src/pipeline.py
import time
import cv2
from .detector import PlateDetector
from .preprocessor import PlatePreprocessor
from .recognizer import PlateRecognizer
from .postprocessor import PlatePostprocessor

class LPRPipeline:
    def __init__(self, config):
        self.detector = PlateDetector(
            config['detector']['model'],
            config['detector']['conf_threshold']
        )
        self.preprocessor = PlatePreprocessor(config.get('preprocess_method', 'standard'))
        self.recognizer = PlateRecognizer(gpu=config.get('gpu', True))
        self.postprocessor = PlatePostprocessor()

    def process(self, image, return_timing=False):
        timings = {}
        results = []

        # 1. 검출
        t = time.time()
        detections = self.detector.detect(image)
        timings['detection'] = time.time() - t

        for det in detections:
            x1, y1, x2, y2 = det['bbox']
            cropped = image[y1:y2, x1:x2]
            if cropped.size == 0:
                continue

            # 2. 전처리
            t = time.time()
            processed = self.preprocessor.process(cropped)
            timings['preprocess'] = timings.get('preprocess', 0) + (time.time() - t)

            # 3. OCR
            t = time.time()
            raw_text, ocr_conf = self.recognizer.recognize(processed)
            timings['ocr'] = timings.get('ocr', 0) + (time.time() - t)

            # 4. 후처리
            t = time.time()
            final_text, is_valid, plate_type = self.postprocessor.process(raw_text)
            timings['postprocess'] = timings.get('postprocess', 0) + (time.time() - t)

            results.append({
                'bbox': det['bbox'],
                'det_confidence': det['confidence'],
                'raw_text': raw_text,
                'final_text': final_text,
                'ocr_confidence': ocr_conf,
                'is_valid': is_valid,
                'plate_type': plate_type,
            })

        if return_timing:
            return results, timings
        return results
```

```python
# 데모용 결과 비디오 생성
def create_demo_video(pipeline, video_path, output_path):
    """LPR 데모 비디오 생성"""
    cap = cv2.VideoCapture(video_path)
    fps = cap.get(cv2.CAP_PROP_FPS)
    w = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    h = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))

    fourcc = cv2.VideoWriter_fourcc(*'mp4v')
    out = cv2.VideoWriter(output_path, fourcc, fps, (w, h))

    frame_count = 0
    while cap.isOpened():
        ret, frame = cap.read()
        if not ret:
            break

        results = pipeline.process(frame)

        # 결과 시각화
        for r in results:
            x1, y1, x2, y2 = r['bbox']
            color = (0, 255, 0) if r['is_valid'] else (0, 0, 255)
            cv2.rectangle(frame, (x1, y1), (x2, y2), color, 2)

            label = f"{r['final_text']} ({r['ocr_confidence']:.0%})"
            cv2.putText(frame, label, (x1, y1-10),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.7, color, 2)

        out.write(frame)
        frame_count += 1

    cap.release()
    out.release()
    print(f"데모 비디오 저장: {output_path} ({frame_count} 프레임)")
```
