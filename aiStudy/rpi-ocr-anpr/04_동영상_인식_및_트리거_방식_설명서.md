## 1. 인식 방식 개요

### 1.1 두 가지 인식 방식

```
┌──────────────────────────────────────────────────────────────────┐
│                      인식 방식 비교                               │
├──────────────────────────────────────────────────────────────────┤
│                                                                  │
│  [방식 A] 연속 자동 인식                                         │
│  ──────────────────────                                          │
│  - 동영상의 모든 프레임(또는 N번째 프레임)을 자동으로 처리       │
│  - 번호판이 감지되면 즉시 인식 시도                              │
│  - 사용자 개입 없이 계속 동작                                    │
│                                                                  │
│  [방식 B] 트리거 기반 인식                                       │
│  ──────────────────────                                          │
│  - 특정 이벤트 발생 시에만 인식 수행                             │
│  - 키보드, 버튼, 센서, 모션 감지 등이 트리거                     │
│  - 사용자나 외부 신호가 인식 시점 결정                           │
│                                                                  │
└──────────────────────────────────────────────────────────────────┘
```

### 1.2 각 방식의 특징

| 항목 | 방식 A (연속) | 방식 B (트리거) |
|------|--------------|----------------|
| 인식 시점 | 자동/연속 | 이벤트 발생 시 |
| CPU 사용량 | 높음 | 낮음 (대기 시) |
| 실시간성 | 즉각적 | 트리거에 의존 |
| 사용 사례 | 교통 모니터링 | 주차장 입출차 |
| 복잡도 | 중복 처리 필요 | 단순 |

```python
import time

class DuplicateFilter:
    def __init__(self, timeout=5):
        """
        timeout: 동일 번호판 재인식 방지 시간 (초)
        """
        self.recent_plates = {}  # {번호판: 마지막_인식_시간}
        self.timeout = timeout

    def is_new(self, plate_number):
        """새로운 번호판인지 확인"""
        current_time = time.time()

        # 오래된 기록 삭제
        expired = [k for k, v in self.recent_plates.items()
                   if current_time - v > self.timeout]
        for k in expired:
            del self.recent_plates[k]

        # 중복 확인
        if plate_number in self.recent_plates:
            return False  # 중복 (최근에 인식됨)

        # 새 번호판 기록
        self.recent_plates[plate_number] = current_time
        return True  # 신규

# 사용 예
filter = DuplicateFilter(timeout=5)

for plate in recognized_plates:
    if filter.is_new(plate):
        print(f"새 번호판 감지: {plate}")
        save_to_log(plate)
    else:
        # 중복이므로 무시
        pass
```

#### 2.2.3 연속 인식 + 표시 전체 코드
```python
#!/usr/bin/env python3
"""방식 A: 연속 자동 인식 시스템"""

import cv2
import time
from collections import deque

class ContinuousRecognizer:
    def __init__(self, camera):
        self.camera = camera
        self.frame_skip = 3
        self.duplicate_timeout = 5
        self.recent_plates = {}

        # 최근 인식 결과 (화면 표시용)
        self.display_results = deque(maxlen=5)

    def run(self):
        """메인 루프"""
        frame_count = 0
        fps_start = time.time()
        fps = 0

        print("연속 인식 시작 - 'q'로 종료")

        while True:
            ret, frame = self.camera.read()
            if not ret:
                break

            frame_count += 1

            # FPS 계산
            if time.time() - fps_start >= 1:
                fps = frame_count
                frame_count = 0
                fps_start = time.time()

            # N번째 프레임에서만 인식 수행
            if frame_count % self.frame_skip == 0:
                self._process_and_recognize(frame)

            # 화면에 결과 표시
            display_frame = self._draw_results(frame, fps)
            cv2.imshow('Continuous Recognition', display_frame)

            if cv2.waitKey(1) & 0xFF == ord('q'):
                break

        cv2.destroyAllWindows()

    def _process_and_recognize(self, frame):
        """검출 + 인식 수행"""
        # 전처리
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        blur = cv2.GaussianBlur(gray, (5, 5), 0)
        edges = cv2.Canny(blur, 100, 200)

        # 번호판 후보 검출
        contours, _ = cv2.findContours(edges, cv2.RETR_TREE, cv2.CHAIN_APPROX_SIMPLE)

        for contour in contours:
            area = cv2.contourArea(contour)
            if area < 500 or area > 50000:
                continue

            x, y, w, h = cv2.boundingRect(contour)
            aspect_ratio = w / h

            if 2.0 < aspect_ratio < 6.0:
                # ROI 추출 및 OCR
                roi = frame[y:y+h, x:x+w]
                text = self._recognize_plate(roi)

                if text and self._is_new_plate(text):
                    # 새 번호판 발견
                    result = {
                        'text': text,
                        'bbox': (x, y, w, h),
                        'time': time.strftime('%H:%M:%S')
                    }
                    self.display_results.append(result)
                    print(f"[{result['time']}] 새 번호판: {text}")

    def _recognize_plate(self, roi):
        """OCR 수행"""
        import pytesseract
        import re

        # 전처리
        gray = cv2.cvtColor(roi, cv2.COLOR_BGR2GRAY) if len(roi.shape) == 3 else roi
        _, binary = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)

        # OCR
        text = pytesseract.image_to_string(binary, lang='kor+eng', config='--psm 7')
        text = re.sub(r'[^0-9가-힣]', '', text)

        return text if len(text) >= 4 else None

    def _is_new_plate(self, plate):
        """중복 확인"""
        current = time.time()

        # 오래된 기록 삭제
        expired = [k for k, v in self.recent_plates.items()
                   if current - v > self.duplicate_timeout]
        for k in expired:
            del self.recent_plates[k]

        if plate in self.recent_plates:
            return False

        self.recent_plates[plate] = current
        return True

    def _draw_results(self, frame, fps):
        """결과 표시"""
        display = frame.copy()

        # FPS 표시
        cv2.putText(display, f"FPS: {fps}", (10, 30),
                   cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 255), 2)

        # 모드 표시
        cv2.putText(display, "Mode: CONTINUOUS", (10, 60),
                   cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)

        # 최근 인식 결과 목록
        y_pos = 100
        cv2.putText(display, "Recent Plates:", (10, y_pos),
                   cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)

        for result in self.display_results:
            y_pos += 25
            text = f"[{result['time']}] {result['text']}"
            cv2.putText(display, text, (10, y_pos),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)

            # 바운딩 박스 그리기
            x, y, w, h = result['bbox']
            cv2.rectangle(display, (x, y), (x+w, y+h), (0, 255, 0), 2)

        return display
```

### 3.2 트리거 1: 키보드 입력

```python
#!/usr/bin/env python3
"""트리거 방식: 키보드 입력으로 촬영+인식"""

import cv2
import time

class KeyboardTriggerRecognizer:
    def __init__(self, camera):
        self.camera = camera
        self.capture_key = ord(' ')  # Space 키
        self.results = []

    def run(self):
        print("=" * 50)
        print("키보드 트리거 모드")
        print("  Space: 촬영 및 인식")
        print("  q: 종료")
        print("=" * 50)

        while True:
            ret, frame = self.camera.read()
            if not ret:
                break

            # 안내 메시지 표시
            display = frame.copy()
            cv2.putText(display, "Press SPACE to capture", (10, 30),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 255), 2)
            cv2.putText(display, "Mode: KEYBOARD TRIGGER", (10, 60),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 165, 255), 2)

            # 최근 결과 표시
            if self.results:
                last = self.results[-1]
                cv2.putText(display, f"Last: {last['text']}", (10, 90),
                           cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)

            cv2.imshow('Keyboard Trigger', display)

            # 키 입력 대기
            key = cv2.waitKey(1) & 0xFF

            if key == self.capture_key:
                # Space 키 → 촬영+인식
                print("\n[트리거] Space 키 입력 - 촬영 시작...")
                self._capture_and_recognize(frame)

            elif key == ord('q'):
                break

        cv2.destroyAllWindows()

    def _capture_and_recognize(self, frame):
        """촬영 및 인식 수행"""
        timestamp = time.strftime('%Y%m%d_%H%M%S')

        # 이미지 저장
        filename = f"capture_{timestamp}.jpg"
        cv2.imwrite(filename, frame)
        print(f"  저장: {filename}")

        # 인식 수행
        text = self._recognize(frame)

        if text:
            result = {
                'text': text,
                'time': timestamp,
                'filename': filename
            }
            self.results.append(result)
            print(f"  인식 결과: {text}")
        else:
            print("  인식 실패: 번호판을 찾을 수 없습니다")

    def _recognize(self, frame):
        """인식 수행 (전체 파이프라인)"""
        import pytesseract
        import re

        # 전처리 및 검출 (간략화)
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        blur = cv2.GaussianBlur(gray, (5, 5), 0)
        edges = cv2.Canny(blur, 100, 200)

        contours, _ = cv2.findContours(edges, cv2.RETR_TREE, cv2.CHAIN_APPROX_SIMPLE)

        for contour in contours:
            area = cv2.contourArea(contour)
            if area < 500:
                continue

            x, y, w, h = cv2.boundingRect(contour)
            if 2.0 < w/h < 6.0:
                roi = frame[y:y+h, x:x+w]
                roi_gray = cv2.cvtColor(roi, cv2.COLOR_BGR2GRAY)
                _, binary = cv2.threshold(roi_gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)

                text = pytesseract.image_to_string(binary, lang='kor+eng', config='--psm 7')
                text = re.sub(r'[^0-9가-힣]', '', text)

                if len(text) >= 4:
                    return text

        return None
```

```python
    GPIO_AVAILABLE = False

class GPIOButtonTrigger:
    def __init__(self, camera, button_pin=17):
        self.camera = camera
        self.button_pin = button_pin
        self.capture_flag = False

        if GPIO_AVAILABLE:
            self._setup_gpio()

    def _setup_gpio(self):
        """GPIO 설정"""
        GPIO.setmode(GPIO.BCM)
        GPIO.setup(self.button_pin, GPIO.IN, pull_up_down=GPIO.PUD_UP)

        # 인터럽트 설정 (버튼 누름 감지)
        GPIO.add_event_detect(
            self.button_pin,
            GPIO.FALLING,  # HIGH → LOW (버튼 누름)
            callback=self._button_callback,
            bouncetime=300  # 디바운싱 (300ms)
        )
        print(f"GPIO {self.button_pin}번 핀에 버튼 설정 완료")

        """버튼 눌림 콜백"""
```

### 3.4 트리거 3: 모션 감지

```python
#!/usr/bin/env python3
"""트리거 방식: 모션 감지로 자동 촬영+인식"""

import cv2
import numpy as np
import time

class MotionTriggerRecognizer:
    def __init__(self, camera, sensitivity=5000):
        self.camera = camera
        self.sensitivity = sensitivity  # 움직임 감지 임계값
        self.prev_frame = None
        self.cooldown = 3  # 연속 트리거 방지 (초)
        self.last_trigger = 0

    def run(self):
        print("=" * 50)
        print("모션 감지 트리거 모드")
        print(f"  감도: {self.sensitivity}")
        print("  q: 종료")
        print("=" * 50)

        while True:
            ret, frame = self.camera.read()
            if not ret:
                break

            # 모션 감지
            motion_detected, diff_value = self._detect_motion(frame)

            # 쿨다운 확인
            current_time = time.time()
            can_trigger = (current_time - self.last_trigger) > self.cooldown

            if motion_detected and can_trigger:
                print(f"\n[트리거] 모션 감지! (diff={diff_value})")
                self.last_trigger = current_time
                self._capture_and_recognize(frame)

            # 화면 표시
            display = frame.copy()
            status = "MOTION!" if motion_detected else "Waiting..."
            color = (0, 0, 255) if motion_detected else (0, 255, 0)

            cv2.putText(display, f"Status: {status}", (10, 30),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.7, color, 2)
            cv2.putText(display, f"Diff: {diff_value}", (10, 60),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 255, 255), 2)
            cv2.putText(display, "Mode: MOTION TRIGGER", (10, 90),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 165, 0), 2)

            cv2.imshow('Motion Trigger', display)

            if cv2.waitKey(1) & 0xFF == ord('q'):
                break

        cv2.destroyAllWindows()

    def _detect_motion(self, frame):
        """프레임 간 차이로 모션 감지"""
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        gray = cv2.GaussianBlur(gray, (21, 21), 0)

        if self.prev_frame is None:
            self.prev_frame = gray
            return False, 0

        # 프레임 차이 계산
        diff = cv2.absdiff(self.prev_frame, gray)
        _, thresh = cv2.threshold(diff, 25, 255, cv2.THRESH_BINARY)

        # 변화량 계산
        diff_value = np.sum(thresh) // 255

        # 이전 프레임 업데이트
        self.prev_frame = gray

        return diff_value > self.sensitivity, diff_value

    def _capture_and_recognize(self, frame):
        """촬영 및 인식"""
        # (키보드 트리거와 동일한 로직)
```

## 5. 실전 구현 코드

### 5.1 통합 인식 시스템 (모든 방식 지원)

```python
#!/usr/bin/env python3
"""통합 번호판 인식 시스템 - 다양한 트리거 지원"""

import cv2
import time
import argparse
import pytesseract
import re
import numpy as np

class PlateRecognizer:
    """번호판 인식 엔진"""

    def recognize(self, frame):
        """프레임에서 번호판 인식"""
        # 전처리
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        blur = cv2.GaussianBlur(gray, (5, 5), 0)
        edges = cv2.Canny(blur, 100, 200)

        # 컨투어 검출
        contours, _ = cv2.findContours(edges, cv2.RETR_TREE, cv2.CHAIN_APPROX_SIMPLE)

        results = []
        for contour in contours:
            area = cv2.contourArea(contour)
            if area < 500 or area > 50000:
                continue

            x, y, w, h = cv2.boundingRect(contour)
            if not (2.0 < w/h < 6.0):
                continue

            # ROI 추출 및 OCR
            roi = frame[y:y+h, x:x+w]
            text, conf = self._ocr(roi)

            if text and len(text) >= 4:
                results.append({
                    'text': text,
                    'confidence': conf,
                    'bbox': (x, y, w, h),
                    'time': time.strftime('%H:%M:%S')
                })

        return results

    def _ocr(self, roi):
        """OCR 수행"""
        gray = cv2.cvtColor(roi, cv2.COLOR_BGR2GRAY) if len(roi.shape) == 3 else roi
        _, binary = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)

        if np.mean(binary) < 127:
            binary = cv2.bitwise_not(binary)

        # 스케일업
        binary = cv2.resize(binary, None, fx=2, fy=2)
        binary = cv2.copyMakeBorder(binary, 10, 10, 10, 10, cv2.BORDER_CONSTANT, value=255)

        # 인식
        data = pytesseract.image_to_data(binary, lang='kor+eng', config='--psm 7',
                                         output_type=pytesseract.Output.DICT)

        text = pytesseract.image_to_string(binary, lang='kor+eng', config='--psm 7')
        text = re.sub(r'[^0-9가-힣]', '', text)

        # 신뢰도
        confs = [int(c) for c in data['conf'] if int(c) > 0]
        avg_conf = sum(confs) / len(confs) if confs else 0

        return text, avg_conf


class IntegratedSystem:
    """통합 인식 시스템"""

    def __init__(self, mode='continuous'):
        self.mode = mode
        self.recognizer = PlateRecognizer()
        self.recent_plates = {}
        self.duplicate_timeout = 5
        self.results_history = []

        # 카메라 초기화
        self.camera = cv2.VideoCapture(0)
        self.camera.set(cv2.CAP_PROP_FRAME_WIDTH, 640)
        self.camera.set(cv2.CAP_PROP_FRAME_HEIGHT, 480)

        # 모션 감지용
        self.prev_frame = None
        self.motion_threshold = 5000

    def run(self):
        """메인 실행"""
        print(f"\n{'='*50}")
        print(f"번호판 인식 시스템 - 모드: {self.mode.upper()}")
        print(f"{'='*50}")
        print("  Space: 수동 촬영 (모든 모드)")
        print("  m: 모드 변경")
        print("  q: 종료\n")

        frame_count = 0
        fps = 0
        fps_start = time.time()

        while True:
            ret, frame = self.camera.read()
            if not ret:
                break

            frame_count += 1
            current_result = None

            # FPS 계산
            if time.time() - fps_start >= 1:
                fps = frame_count
                frame_count = 0
                fps_start = time.time()

            # 모드별 처리
            if self.mode == 'continuous':
                # 3프레임마다 인식
                if frame_count % 3 == 0:
                    results = self.recognizer.recognize(frame)
                    for r in results:
                        if self._is_new(r['text']):
                            current_result = r
                            self._log_result(r)

            elif self.mode == 'motion':
                if self._detect_motion(frame):
                    results = self.recognizer.recognize(frame)
                    for r in results:
                        if self._is_new(r['text']):
                            current_result = r
                            self._log_result(r)

            # 화면 표시
            display = self._render_display(frame, current_result, fps)
            cv2.imshow('Plate Recognition', display)

            # 키 입력 처리
            key = cv2.waitKey(1) & 0xFF

            if key == ord(' '):  # 수동 촬영
                print("\n[수동] 촬영...")
                results = self.recognizer.recognize(frame)
                for r in results:
                    self._log_result(r)
                    print(f"  인식: {r['text']} ({r['confidence']:.0f}%)")

            elif key == ord('m'):  # 모드 변경
                modes = ['continuous', 'motion', 'manual']
                idx = modes.index(self.mode)
                self.mode = modes[(idx + 1) % len(modes)]
                print(f"\n모드 변경: {self.mode.upper()}")

            elif key == ord('q'):
                break

        self.camera.release()
        cv2.destroyAllWindows()

    def _is_new(self, plate):
        """중복 확인"""
        now = time.time()
        expired = [k for k, v in self.recent_plates.items() if now - v > self.duplicate_timeout]
        for k in expired:
            del self.recent_plates[k]

        if plate in self.recent_plates:
            return False
        self.recent_plates[plate] = now
        return True

    def _detect_motion(self, frame):
        """모션 감지"""
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        gray = cv2.GaussianBlur(gray, (21, 21), 0)

        if self.prev_frame is None:
            self.prev_frame = gray
            return False

        diff = cv2.absdiff(self.prev_frame, gray)
        _, thresh = cv2.threshold(diff, 25, 255, cv2.THRESH_BINARY)
        motion = np.sum(thresh) // 255 > self.motion_threshold

        self.prev_frame = gray
        return motion

    def _log_result(self, result):
        """결과 기록"""
        self.results_history.insert(0, result)
        if len(self.results_history) > 10:
            self.results_history.pop()
        print(f"[{result['time']}] {result['text']} ({result['confidence']:.0f}%)")

    def _render_display(self, frame, current_result, fps):
        """화면 렌더링"""
        display = frame.copy()
        h, w = display.shape[:2]

        # 상단 바
        cv2.rectangle(display, (0, 0), (w, 60), (30, 30, 30), -1)
        cv2.putText(display, f"FPS: {fps}", (10, 25),
                   cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 255), 2)

        mode_colors = {'continuous': (0, 255, 0), 'motion': (255, 165, 0), 'manual': (255, 0, 255)}
        cv2.putText(display, f"Mode: {self.mode.upper()}", (w-180, 25),
                   cv2.FONT_HERSHEY_SIMPLEX, 0.6, mode_colors.get(self.mode, (255,255,255)), 2)

        if current_result:
            cv2.putText(display, f"Detected: {current_result['text']}", (10, 50),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)

            # 바운딩 박스
            x, y, bw, bh = current_result['bbox']
            cv2.rectangle(display, (x, y), (x+bw, y+bh), (0, 255, 0), 2)

        # 하단 히스토리
        cv2.rectangle(display, (0, h-100), (w, h), (40, 40, 40), -1)
        cv2.putText(display, "History:", (10, h-80),
                   cv2.FONT_HERSHEY_SIMPLEX, 0.4, (200, 200, 200), 1)

        for i, r in enumerate(self.results_history[:4]):
            y_pos = h - 60 + i * 15
            cv2.putText(display, f"[{r['time']}] {r['text']}", (10, y_pos),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.35, (0, 255, 0), 1)

        return display


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument('--mode', choices=['continuous', 'motion', 'manual'],
                       default='continuous', help='인식 모드')
    args = parser.parse_args()

    system = IntegratedSystem(mode=args.mode)
    system.run()
```
