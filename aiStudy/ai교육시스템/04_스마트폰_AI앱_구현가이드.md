```kotlin
class MainActivity : AppCompatActivity() {

    private lateinit var cameraProvider: ProcessCameraProvider
    private lateinit var preview: Preview
    private lateinit var imageCapture: ImageCapture

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        // 카메라 권한 확인
        if (checkCameraPermission()) {
            startCamera()
        } else {
            requestCameraPermission()
        }

        // 캡처 버튼 클릭
        captureButton.setOnClickListener {
            takePhoto()
        }
    }

    private fun startCamera() {
        val cameraProviderFuture = ProcessCameraProvider.getInstance(this)

        cameraProviderFuture.addListener({
            cameraProvider = cameraProviderFuture.get()

            preview = Preview.Builder().build().also {
                it.setSurfaceProvider(previewView.surfaceProvider)
            }

            imageCapture = ImageCapture.Builder().build()

            val cameraSelector = CameraSelector.DEFAULT_BACK_CAMERA

            cameraProvider.unbindAll()
            cameraProvider.bindToLifecycle(
                this, cameraSelector, preview, imageCapture
            )

        }, ContextCompat.getMainExecutor(this))
    }
}
```

```kotlin
private fun takePhoto() {
    val outputOptions = ImageCapture.OutputFileOptions.Builder(
        File(cacheDir, "captured.jpg")
    ).build()

    imageCapture.takePicture(
        outputOptions,
        ContextCompat.getMainExecutor(this),
        object : ImageCapture.OnImageSavedCallback {
            override fun onImageSaved(output: ImageCapture.OutputFileResults) {
                val savedUri = output.savedUri
                recognizeText(savedUri)
            }

            override fun onError(exception: ImageCaptureException) {
                Log.e("Capture", "캡처 실패: ${exception.message}")
            }
        }
    )
```

#### Step 8: ML Kit 연동 (핵심!)

**MIT App Inventor:**
```
┌─────────────────────────────────────────────────────────────────┐
│  [TextRecognizer1.GotResult] (결과텍스트)                       │
│      └─ [ResultTextBox.Text ← 결과텍스트]                       │
```

**Android Studio (핵심 코드):**
```kotlin
import com.google.mlkit.vision.text.TextRecognition
import com.google.mlkit.vision.text.korean.KoreanTextRecognizerOptions

private fun recognizeText(imageUri: Uri?) {
    if (imageUri == null) return

    // 이미지 로드
    val image = InputImage.fromFilePath(this, imageUri)

    // 한국어 텍스트 인식기 생성
    val recognizer = TextRecognition.getClient(
        KoreanTextRecognizerOptions.Builder().build()
    )

    // 인식 실행
    recognizer.process(image)
        .addOnSuccessListener { result ->
            // 성공: 결과 처리
            processResult(result)
        }
        .addOnFailureListener { e ->
            // 실패: 에러 처리
            resultText.text = "인식 실패: ${e.message}"
        }
}

private fun processResult(result: Text) {
    val recognizedText = result.text

    if (recognizedText.isEmpty()) {
        resultText.text = "텍스트를 찾을 수 없습니다"
        return
    }

    // 결과 표시
    resultText.text = recognizedText

    // 번호판 패턴 추출 (한국 번호판: 12가3456)
    val platePattern = Regex("\\d{2,3}[가-힣]\\s?\\d{4}")
    val plateMatch = platePattern.find(recognizedText)

    if (plateMatch != null) {
        resultText.text = "🚗 ${plateMatch.value}"
        confidenceText.text = "신뢰도: 높음 ✅"
    }

    // 블록별 신뢰도 계산
    var totalConfidence = 0f
    var blockCount = 0

    for (block in result.textBlocks) {
        for (line in block.lines) {
            line.confidence?.let {
                totalConfidence += it
                blockCount++
            }
        }
    }

    if (blockCount > 0) {
        val avgConfidence = (totalConfidence / blockCount * 100).toInt()
        confidenceText.text = "신뢰도: ${avgConfidence}%"

        // 색상 변경
        val color = when {
            avgConfidence >= 90 -> Color.GREEN
            avgConfidence >= 70 -> Color.YELLOW
            else -> Color.RED
        }
        confidenceText.setTextColor(color)
    }
}
```
