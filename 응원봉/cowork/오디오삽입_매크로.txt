Sub InsertAudioToSlides()
    '
    ' PPT 각 슬라이드에 나레이션 MP3 자동 삽입 + 자동 슬라이드 전환
    ' audio_files 폴더의 slide1.mp3 ~ slide22.mp3를 삽입합니다
    '
    ' [기능]
    ' 1. 각 슬라이드에 MP3 오디오 삽입
    ' 2. 슬라이드 시작 시 자동 재생
    ' 3. 오디오 끝나면 자동으로 다음 슬라이드로 전환
    '

    Dim prs As Presentation
    Dim sld As Slide
    Dim shp As Shape
    Dim audioPath As String
    Dim audioFolder As String
    Dim slideNum As Integer
    Dim insertCount As Integer

    Set prs = ActivePresentation

    ' audio_files 폴더 경로 (PPT 파일과 같은 위치)
    audioFolder = prs.Path & "\audio_files\"

    ' 폴더 존재 확인
    If Dir(audioFolder, vbDirectory) = "" Then
        MsgBox "audio_files 폴더를 찾을 수 없습니다!" & vbCrLf & vbCrLf & _
               "PPT 파일과 같은 폴더에 audio_files 폴더가 있어야 합니다." & vbCrLf & _
               "경로: " & audioFolder, vbExclamation, "오류"
        Exit Sub
    End If

    ' 각 슬라이드별 오디오 재생 시간 (초) + 여유 2초
    Dim durations(1 To 22) As Single
    durations(1) = 23
    durations(2) = 25
    durations(3) = 28
    durations(4) = 27
    durations(5) = 35
    durations(6) = 23
    durations(7) = 24
    durations(8) = 26
    durations(9) = 40
    durations(10) = 29
    durations(11) = 31
    durations(12) = 25
    durations(13) = 22
    durations(14) = 33
    durations(15) = 32
    durations(16) = 28
    durations(17) = 22
    durations(18) = 25
    durations(19) = 28
    durations(20) = 31
    durations(21) = 27
    durations(22) = 12

    insertCount = 0

    For Each sld In prs.Slides
        slideNum = sld.SlideIndex

        ' 슬라이드 번호가 22를 넘으면 건너뜀
        If slideNum > 22 Then GoTo NextSlide

        audioPath = audioFolder & "slide" & slideNum & ".mp3"

        ' MP3 파일 존재 확인
        If Dir(audioPath) <> "" Then

            ' 오디오 삽입
            Set shp = sld.Shapes.AddMediaObject2( _
                FileName:=audioPath, _
                LinkToFile:=msoFalse, _
                SaveWithDocument:=msoTrue, _
                Left:=10, _
                Top:=500, _
                Width:=24, _
                Height:=24)

            ' 자동 재생 설정
            With sld.TimeLine.MainSequence.AddEffect( _
                Shape:=shp, _
                effectId:=msoAnimEffectMediaPlay, _
                trigger:=msoAnimTriggerWithPrevious)

                .Timing.TriggerType = msoAnimTriggerWithPrevious
            End With

            ' 슬라이드 자동 전환 설정 (오디오 끝나면 다음 슬라이드)
            With sld.SlideShowTransition
                .AdvanceOnClick = msoTrue          ' 클릭으로도 넘기기 가능
                .AdvanceOnTime = msoTrue            ' 시간 경과 후 자동 전환
                .AdvanceTime = durations(slideNum)  ' 오디오 길이 + 여유시간
                .Speed = ppTransitionSpeedFast      ' 전환 속도
            End With

            insertCount = insertCount + 1
            Debug.Print "Slide " & slideNum & ": OK (" & durations(slideNum) & "s)"
        Else
            Debug.Print "Slide " & slideNum & ": MP3 not found"
        End If

NextSlide:
    Next sld

    MsgBox insertCount & "개 슬라이드에 나레이션이 삽입되었습니다!" & vbCrLf & vbCrLf & _
           "✓ 각 슬라이드에서 오디오가 자동 재생됩니다" & vbCrLf & _
           "✓ 오디오가 끝나면 자동으로 다음 슬라이드로 넘어갑니다" & vbCrLf & vbCrLf & _
           "[다른 이름으로 저장] 후 F5로 슬라이드 쇼를 시작하세요!", _
           vbInformation, "완료"

End Sub
