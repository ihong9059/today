Sub InsertAudioAndAutoPlay()
'
' 오디오 삽입 + 자동재생 한번에 처리
'
' [사용법]
'   1. AI_FanStick_제안서.pptx (음성 없는 원본)을 PowerPoint에서 엽니다
'   2. Alt + F11 (VBA 편집기 열기)
'   3. 삽입(Insert) > 모듈(Module) 클릭
'   4. 이 코드를 전체 복사하여 붙여넣기
'   5. F5 눌러 실행
'   6. Alt + Q (VBA 편집기 닫기)
'   7. Ctrl + Shift + S 로 다른 이름으로 저장
'   8. F5 눌러 슬라이드쇼에서 소리 확인!
'

    Dim sld As Slide
    Dim shp As Shape
    Dim audioPath As String
    Dim basePath As String
    Dim i As Long
    Dim audioCount As Long

    basePath = ActivePresentation.Path & "\audio_files\"
    audioCount = 0

    ' audio_files 폴더 확인
    If Dir(basePath, vbDirectory) = "" Then
        MsgBox "audio_files 폴더를 찾을 수 없습니다!" & vbCrLf & vbCrLf & _
               "확인 경로: " & basePath & vbCrLf & vbCrLf & _
               "PPT 파일과 같은 폴더에 audio_files 폴더가 있어야 합니다.", _
               vbExclamation, "오류"
        Exit Sub
    End If

    For i = 1 To ActivePresentation.Slides.Count
        Set sld = ActivePresentation.Slides(i)
        audioPath = basePath & "slide" & i & ".mp3"

        ' MP3 파일 존재 확인
        If Dir(audioPath) <> "" Then

            ' 오디오 삽입
            Set shp = sld.Shapes.AddMediaObject2( _
                FileName:=audioPath, _
                LinkToFile:=msoFalse, _
                SaveWithDocument:=msoTrue, _
                Left:=10, Top:=500, Width:=24, Height:=24)

            ' 자동재생 설정
            shp.AnimationSettings.PlaySettings.PlayOnEntry = True
            shp.AnimationSettings.PlaySettings.HideWhileNotPlaying = True
            shp.AnimationSettings.PlaySettings.StopAfterSlides = 1
            shp.AnimationSettings.Animate = True

            audioCount = audioCount + 1
        End If
    Next i

    MsgBox audioCount & "개 슬라이드에 오디오를 삽입하고 자동재생을 설정했습니다!" & vbCrLf & vbCrLf & _
           "1. Alt+Q 로 이 창을 닫으세요" & vbCrLf & _
           "2. Ctrl+Shift+S 로 다른 이름으로 저장" & vbCrLf & _
           "3. F5 로 슬라이드쇼에서 확인!", _
           vbInformation, "완료"

End Sub
