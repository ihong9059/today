

```bash
# 이미지 개수
```

## 4. PM2 관리

```bash
# 상태 확인
ssh uttec-ec2 "pm2 list"

# 전체 재시작
ssh uttec-ec2 "pm2 restart all"

# 개별 재시작
ssh uttec-ec2 "pm2 restart sensor-monitor"
ssh uttec-ec2 "pm2 restart cert-guide"

# 중지
ssh uttec-ec2 "pm2 stop sensor-monitor"
ssh uttec-ec2 "pm2 stop cert-guide"

# 시작
ssh uttec-ec2 "pm2 start sensor-monitor"
ssh uttec-ec2 "pm2 start cert-guide"

# 로그 정리
ssh uttec-ec2 "pm2 flush"

# 상태 저장
ssh uttec-ec2 "pm2 save"
```

## 5. 데이터베이스 (PostgreSQL)

```bash
# 데이터 개수 확인
ssh uttec-ec2 "psql -U sensoruser -d sensordb -c 'SELECT COUNT(*) FROM sensor_data'"

# 디바이스별 데이터 수
ssh uttec-ec2 "psql -U sensoruser -d sensordb -c 'SELECT device, COUNT(*) FROM sensor_data GROUP BY device'"

# 최근 데이터 10건
ssh uttec-ec2 "psql -U sensoruser -d sensordb -c 'SELECT id, device, timestamp FROM sensor_data ORDER BY id DESC LIMIT 10'"

# 7일 이전 데이터 삭제
ssh uttec-ec2 "psql -U sensoruser -d sensordb -c \"DELETE FROM sensor_data WHERE created_at < NOW() - INTERVAL '7 days'\""
```

## 6. 문제 해결

### 서비스 안될 때

```bash
# 1. PM2 상태 확인
ssh uttec-ec2 "pm2 list"

# 2. 에러 로그 확인
ssh uttec-ec2 "pm2 logs --err --lines 100"

# 3. 프로세스 재시작
ssh uttec-ec2 "pm2 restart all"

# 4. 그래도 안되면 삭제 후 재시작
ssh uttec-ec2 "pm2 delete all"
ssh uttec-ec2 "cd ~/sensor-monitor && pm2 start server.js --name sensor-monitor"
ssh uttec-ec2 "cd ~/cert-guide && pm2 start npm --name cert-guide -- start"
ssh uttec-ec2 "pm2 save"
```

### 디스크 부족

```bash
# 용량 확인
ssh uttec-ec2 "df -h"

# 큰 폴더 찾기
ssh uttec-ec2 "du -sh ~/* | sort -hr"

# 로그 정리
ssh uttec-ec2 "pm2 flush"

# 이미지 정리
ssh uttec-ec2 "rm -rf ~/sensor-monitor/images/*"

# npm 캐시 정리
ssh uttec-ec2 "npm cache clean --force"
```

### 접속 안될 때

```bash
# 포트 확인
nc -zv 52.78.119.132 22
nc -zv 52.78.119.132 5000
nc -zv 52.78.119.132 3000

# SSH 상세 로그
ssh -v uttec-ec2
```
